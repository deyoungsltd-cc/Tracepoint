    64→    64→function formatTimestamp(ts: string | null): string {
    65→    65→  if (!ts) return '—';
    66→    66→  const d = new Date(ts);
    67→    67→  const now = new Date();
    68→    68→  const diffMs = now.getTime() - d.getTime();
    69→    69→  const diffMin = Math.floor(diffMs / 60000);
    70→    70→  const diffHr = Math.floor(diffMs / 3600000);
    71→    71→  const diffDay = Math.floor(diffMs / 86400000);
    72→    72→
    73→    73→  if (diffMin < 1) return 'Just now';
    74→    74→  if (diffMin < 60) return `${diffMin}m ago`;
    75→    75→  if (diffHr < 24) return `${diffHr}h ago`;
    76→    76→  if (diffDay < 7) return `${diffDay}d ago`;
    77→    77→  return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' });
    78→    78→}
    79→    79→
    80→    80→function formatDuration(start: string | null, end: string | null): string {
    81→    81→  if (!start || !end) return '—';
    82→    82→  const ms = new Date(end).getTime() - new Date(start).getTime();
    83→    83→  const sec = Math.round(ms / 1000);
    84→    84→  if (sec < 60) return `${sec}s`;
    85→    85→  const min = Math.floor(sec / 60);
    86→    86→  const remSec = sec % 60;
    87→    87→  return `${min}m ${remSec}s`;
    88→    88→}

    93→    93→    month: 'short',

    96→    96→    hour: '2-digit',
    97→    97→    minute: '2-digit',
    98→    98→    second: '2-digit',
    99→    99→    hour12: false,
   100→   100→  });
   101→   101→}

   272→   272→function ConfidenceRing({ score, size = 72 }: { score: number; size?: number }) {
   273→   273→  const r = (size - 8) / 2;
   274→   274→  const circ = 2 * Math.PI * r;
   275→   275→  const offset = circ - (score / 100) * circ;
   276→   276→  const color = confidenceColor(score);
   277→   277→
   278→   278→  return (
   279→   279→    <div className="relative" style={{ width: size, height: size }}>
   280→   280→      <svg width={size} height={size} className="tp-confidence-ring">
   281→   281→        <circle
   282→   282→          cx={size / 2}
   283→   283→          cy={size / 2}
   284→   284→          r={r}
   285→   285→          fill="none"
   286→   286→          stroke="#2a3322"
   287→   287→          strokeWidth={4}
   288→   288→        />
   289→   289→        <circle
   290→   290→          cx={size / 2}
   291→   291→          cy={size / 2}
   292→   292→          r={r}
   293→   293→          fill="none"
   294→   294→          stroke={color}
   295→   295→          strokeWidth={4}
   296→   296→          strokeDasharray={circ}
   297→   297→          strokeDashoffset={offset}
   298→   298→          strokeLinecap="round"
   299→   299→          style={{ transition: 'stroke-dashoffset 0.6s ease' }}
   300→   300→        />

   303→   303→        <span

   312→   312→}
