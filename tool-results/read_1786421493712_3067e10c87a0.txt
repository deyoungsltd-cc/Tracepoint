    17→    17→function evidenceReliability(sourceType: string): number {
    18→    18→  const map: Record<string, number> = {

    26→    26→    phone_validation: 95,
    27→    27→    public_record: 88,
    28→    28→  };
    29→    29→  return map[sourceType] || 50;

    42→    42→async function proxyNumVerify(phone: string): Promise<Record<string, unknown> | null> {
    43→    43→  try {
    44→    44→    const res = await fetch(`/api/numverify?phone=${encodeURIComponent(phone)}`);
    45→    45→    if (!res.ok) return null;
    46→    46→    return await res.json();
    47→    47→  } catch { return null; }
    48→    48→}
    49→    49→
    50→    50→async function proxySerperSearch(query: string): Promise<Array<{ title: string; link: string; snippet: string }>> {
    51→    51→  try {
    52→    52→    const res = await fetch('/api/serper', {
    53→    53→      method: 'POST',
    54→    54→      headers: { 'Content-Type': 'application/json' },
    55→    55→      body: JSON.stringify({ query }),
    56→    56→    });
    57→    57→    if (!res.ok) return [];
    58→    58→    const data = await res.json();
    59→    59→    return (data.organic || []).map((item: Record<string, unknown>, i: number) => ({
    60→    60→      title: String(item.title || ''),
    61→    61→      link: String(item.link || ''),
    62→    62→      snippet: String(item.snippet || ''),
    63→    63→    }));
    64→    64→  } catch { return []; }

   309→   309→// Stage 3: Web search via Serper.dev (server proxy)
   310→   310→async function stageWebSearch(config: PipelineConfig, country: string): Promise<{
   311→   311→  searchResults: Array<{ title: string; link: string; snippet: string }>;
   312→   312→  evidence: EvidenceItem[];
   313→   313→  timeline: TimelineEvent[];
   314→   314→}> {
   315→   315→  const evidence: EvidenceItem[] = [];
   316→   316→  const timeline: TimelineEvent[] = [];
   317→   317→  let searchResults: Array<{ title: string; link: string; snippet: string }> = [];
   318→   318→
   319→   319→  const queries: string[] = [];
   320→   320→  if (config.phoneNormalized) queries.push(`"${config.phoneNormalized}"`);
   321→   321→  if (config.email) queries.push(`"${config.email}"`);
   322→   322→  if (config.phoneNormalized && country) {
   323→   323→    queries.push(`${config.phoneNormalized.replace(/[^0-9+]/g, '')} site:linkedin.com OR site:facebook.com OR site:twitter.com`);
   324→   324→  }
   325→   325→
   326→   326→  const maxQueries = config.depth === 'quick' ? 1 : config.depth === 'standard' ? 2 : queries.length;
   327→   327→
   328→   328→  for (let i = 0; i < Math.min(maxQueries, queries.length); i++) {
   329→   329→    const results = await proxySerperSearch(queries[i]);
   330→   330→    searchResults.push(...results);
   331→   331→    timeline.push({
   332→   332→      id: uuid(),
   333→   333→      eventType: 'web_search',
   334→   334→      description: `Search "${queries[i].substring(0, 60)}..." returned ${results.length} results`,
   335→   335→      metadata: { query: queries[i], count: results.length },
   336→   336→      timestamp: timestampNow(),
   337→   337→    });
   338→   338→  }
   339→   339→
   340→   340→  for (const result of searchResults) {
   341→   341→    const relScore = clampScore(
   342→   342→      50 + (config.email && result.snippet.toLowerCase().includes(config.email.toLowerCase()) ? 30 : 0) +
   343→   343→      (config.phoneNormalized && result.snippet.toLowerCase().includes(config.phoneNormalized.replace(/[^0-9]/g, '')) ? 25 : 0)
   344→   344→    );
   345→   345→
   346→   346→    evidence.push({
   347→   347→      id: uuid(),
   348→   348→      claim: result.title,
   349→   349→      sourceName: (() => { try { return new URL(result.link).hostname.replace('www.', ''); } catch { return 'unknown'; } })(),
   350→   350→      sourceType: 'web_search',
   351→   351→      sourceUrl: result.link,
   352→   352→      discoveredAt: timestampNow(),
   353→   353→      publishedAt: null,
   354→   354→      excerpt: result.snippet,
   355→   355→      reliabilityScore: evidenceReliability('web_search'),
   356→   356→      relevanceScore: relScore,
   357→   357→      freshnessScore: 50,
   358→   358→      verificationStatus: relScore >= 70 ? 'possible' : 'unverified',
   359→   359→    });
   360→   360→  }
   361→   361→
   362→   362→  return { searchResults, evidence, timeline };
   363→   363→}

   365→   365→// Stage 4: Correlate candidates from search results
   366→   366→function stageCorrelation(
   367→   367→  evidence: EvidenceItem[],
   368→   368→  phoneInfo: Record<string, unknown> | null,
   369→   369→  config: PipelineConfig
   370→   370→): { candidates: IdentityCandidate[]; updatedEvidence: EvidenceItem[] } {
   371→   371→  const candidateMap = new Map<string, IdentityCandidate>();
   372→   372→
   373→   373→  for (const ev of evidence) {
   374→   374→    if (ev.sourceType === 'phone_validation') continue;
   375→   375→
   376→   376→    let identityKey = '';
   377→   377→    let name: string | null = null;
   378→   378→    const title = ev.claim || '';
   379→   379→    const snippet = ev.excerpt || '';
   380→   380→
   381→   381→    const nameMatch = title.match(/^([A-Z][a-z]+ [A-Z][a-z]+)/);
   382→   382→    if (nameMatch) {
   383→   383→      name = nameMatch[1];
   384→   384→      identityKey = name.toLowerCase();
   385→   385→    } else if (ev.sourceName && !['google', 'bing', 'yahoo'].includes(ev.sourceName)) {
   386→   386→      identityKey = ev.sourceName;
   387→   387→    }
   388→   388→
   389→   389→    if (!identityKey) continue;
   390→   390→
   391→   391→    if (!candidateMap.has(identityKey)) {
   392→   392→      const matchFields: string[] = [];
   393→   393→      if (config.phoneNormalized && snippet.includes(config.phoneNormalized.replace(/[^0-9]/g, ''))) matchFields.push('phone');
   394→   394→      if (config.email && snippet.toLowerCase().includes(config.email.toLowerCase())) matchFields.push('email');
   395→   395→
   396→   396→      candidateMap.set(identityKey, {
   397→   397→        id: uuid(),
   398→   398→        rank: candidateMap.size + 1,
   399→   399→        name,
   400→   400→        phone: config.phoneNormalized || null,
   401→   401→        email: config.email || null,
   402→   402→        business: null,
   403→   403→        website: ev.sourceUrl && ev.sourceUrl.startsWith('http') ? ev.sourceUrl : null,
   404→   404→        location: phoneInfo && phoneInfo.location ? String(phoneInfo.location) : null,
   405→   405→        photoUrl: null,
   406→   406→        confidence: matchFields.length > 0 ? 60 : 30,
   407→   407→        verifiedStatus: matchFields.length >= 2 ? 'possible' : 'unverified',
   408→   408→        matchFields,
   409→   409→        evidence: [],
   410→   410→      });
   411→   411→    }
   412→   412→
   413→   413→    const candidate = candidateMap.get(identityKey)!;
   414→   414→    ev.candidateId = candidate.id;
   415→   415→    candidate.evidence.push(ev);
   416→   416→  }
   417→   417→
   418→   418→  if (candidateMap.size === 0) {
   419→   419→    const phoneEvidence = evidence.filter(e => e.sourceType === 'phone_validation');
   420→   420→    const phoneValid = phoneInfo && phoneInfo.valid;
   421→   421→
   422→   422→    if (phoneValid || config.email) {
   423→   423→      const candidate: IdentityCandidate = {
   424→   424→        id: uuid(),
   425→   425→        rank: 1,
   426→   426→        name: config.email ? config.email.split('@')[0].replace(/[._]/g, ' ').replace(/\b\w/g, c => c.toUpperCase()) : null,
   427→   427→        phone: phoneValid ? String(phoneInfo.international_format || '') : (config.phoneNormalized || null),
   428→   428→        email: config.email || null,
   429→   429→        business: null,
   430→   430→        website: null,
   431→   431→        location: phoneValid ? String(phoneInfo.location || phoneInfo.country_name || '') : null,
   432→   432→        photoUrl: null,
   433→   433→        confidence: phoneValid ? 55 : 35,
   434→   434→        verifiedStatus: phoneValid ? 'possible' : 'unverified',
   435→   435→        matchFields: [...(phoneValid ? ['phone'] : []), ...(config.email ? ['email'] : [])],
   436→   436→        evidence: [],
   437→   437→      };
   438→   438→
   439→   439→      // Link phone validation evidence to this candidate
   440→   440→      for (const ev of phoneEvidence) {
   441→   441→        ev.candidateId = candidate.id;
   442→   442→        candidate.evidence.push(ev);
   443→   443→      }
   444→   444→
   445→   445→      return { candidates: [candidate], updatedEvidence: evidence };
   446→   446→    }
   447→   447→
   448→   448→    return { candidates: [], updatedEvidence: evidence };
   449→   449→  }
   450→   450→
   451→   451→  const sorted = Array.from(candidateMap.values()).sort((a, b) => {
   452→   452→    const aScore = a.confidence * 0.4 + a.evidence.length * 15;
   453→   453→    const bScore = b.confidence * 0.4 + b.evidence.length * 15;
   454→   454→    return bScore - aScore;
   455→   455→  });
   456→   456→  sorted.forEach((c, i) => { c.rank = i + 1; });
   457→   457→
   458→   458→  return { candidates: sorted, updatedEvidence: evidence };
   459→   459→}

   462→   462→async function stageAIAnalysis(
   463→   463→  investigation: Investigation,
   464→   464→): Promise<{ aiAssessment: AIAssessment | null; timeline: TimelineEvent[] }> {
   465→   465→  const timeline: TimelineEvent[] = [];
   466→   466→
   467→   467→  try {
   468→   468→    timeline.push({
   469→   469→      id: uuid(),
   470→   470→      eventType: 'ai_analyzing',
   471→   471→      description: 'AI analyst processing evidence and candidates...',
   472→   472→      metadata: null,
   473→   473→      timestamp: timestampNow(),
   474→   474→    });
   475→   475→
   476→   476→    const assessment = await analyzeIdentity(
   477→   477→      {
   478→   478→        phone: investigation.inputPhoneNormalized || '',
   479→   479→        email: investigation.inputEmail || '',
   480→   480→        candidates: investigation.candidates,
   481→   481→        evidence: investigation.evidence,
   482→   482→        country: investigation.inputCountry || '',
   483→   483→      },
   484→   484→      { model: 'gpt-4o' }
   485→   485→    );
   486→   486→
   487→   487→    if (assessment) {
   488→   488→      timeline.push({
   489→   489→        id: uuid(),
   490→   490→        eventType: 'ai_completed',
   491→   491→        description: `AI assessment complete — ${assessment.confidence.level} confidence (${assessment.confidence.score}%)`,
   492→   492→        metadata: { level: assessment.confidence.level, score: assessment.confidence.score },
   493→   493→        timestamp: timestampNow(),
   494→   494→      });
   495→   495→    }
   496→   496→
   497→   497→    return { aiAssessment: assessment, timeline };
   498→   498→  } catch {
   499→   499→    timeline.push({
   500→   500→      id: uuid(),
   501→   501→      eventType: 'ai_failed',
   502→   502→      description: 'AI analysis failed — continuing without AI assessment',
   503→   503→      metadata: null,
   504→   504→      timestamp: timestampNow(),
   505→   505→    });
   506→   506→    return { aiAssessment: null, timeline };
   507→   507→  }
   508→   508→}

   513→   513→
   514→   514→export async function runRealInvestigation(
   515→   515→  config: PipelineConfig,
   516→   516→  callbacks: PipelineCallbacks
   517→   517→): Promise<{
   518→   518→  investigation: Investigation;
   519→   519→  aiAssessment: AIAssessment | null;
   520→   520→}> {
   521→   521→  const id = uuid();
   522→   522→  const now = timestampNow();
   523→   523→  const allEvidence: EvidenceItem[] = [];
   524→   524→  const allTimeline: TimelineEvent[] = [];
   525→   525→  let country = config.country || '';
   526→   526→
   527→   527→  callbacks.onProgress('initializing', 'Initializing investigation...', 5);
   528→   528→  allTimeline.push({ id: uuid(), eventType: 'started', description: 'Investigation initiated', metadata: { phone: config.phone, email: config.email, depth: config.depth }, timestamp: now });
   529→   529→
   530→   530→  // --- Stage 1: Phone Validation ---
   531→   531→  callbacks.onProgress('validation', 'Validating phone number...', 10);
   532→   532→  const phoneResult = await stagePhoneValidation(config);
   533→   533→  allEvidence.push(...phoneResult.evidence);
   534→   534→  allTimeline.push(...phoneResult.timeline);
   535→   535→  if (phoneResult.country) country = phoneResult.country;
   536→   536→
   537→   537→  // --- Stage 2: Twilio Enrichment ---
   538→   538→  callbacks.onProgress('enrichment', 'Enriching phone data via Twilio...', 22);
   539→   539→  const twilioResult = await stageTwilioEnrichment(config, country, false, null);
   540→   540→  allEvidence.push(...twilioResult.evidence);
   541→   541→  allTimeline.push(...twilioResult.timeline);
   542→   542→  if (twilioResult.country) country = twilioResult.country;
   543→   543→  const twilioCallerName = twilioResult.callerName;
   544→   544→
   545→   545→  // --- Stage 3: Web Search ---
   546→   546→  callbacks.onProgress('discovery', 'Searching public sources...', 38);
   547→   547→  const searchResult = await stageWebSearch(config, country);
   548→   548→  allEvidence.push(...searchResult.evidence);
   549→   549→  allTimeline.push(...searchResult.timeline);
   550→   550→
   551→   551→  // --- Stage 4: Identity Correlation ---
   552→   552→  callbacks.onProgress('correlating', 'Correlating identities...', 58);
   553→   553→  const { candidates, updatedEvidence } = stageCorrelation(allEvidence, phoneResult.phoneInfo, config);
   554→   554→
   555→   555→  // Apply Twilio caller name to candidates that have no name yet
   556→   556→  if (twilioCallerName) {
   557→   557→    for (const c of candidates) {
   558→   558→      if (!c.name) {
   559→   559→        c.name = twilioCallerName;
   560→   560→      }
   561→   561→    }
   562→   562→  }
   563→   563→
   564→   564→  for (const candidate of candidates) {
   565→   565→    const candidateEvidence = allEvidence.filter(e => e.candidateId === candidate.id);
   566→   566→    const baseFromFields = candidate.matchFields.length * 15;
   567→   567→    if (candidateEvidence.length > 0) {
   568→   568→      const avgReliability = candidateEvidence.reduce((sum, e) => sum + e.reliabilityScore, 0) / candidateEvidence.length;
   569→   569→      const avgRelevance = candidateEvidence.reduce((sum, e) => sum + e.relevanceScore, 0) / candidateEvidence.length;
   570→   570→      candidate.confidence = clampScore(avgReliability * 0.4 + avgRelevance * 0.4 + Math.min(candidateEvidence.length * 8, 20) + baseFromFields);
   571→   571→    } else {
   572→   572→      // Even without linked evidence, boost from match fields
   573→   573→      candidate.confidence = clampScore(30 + baseFromFields);
   574→   574→    }
   575→   575→    candidate.verifiedStatus = candidate.confidence >= 80 ? 'verified' : candidate.confidence >= 50 ? 'possible' : 'unverified';
   576→   576→  }
   577→   577→
   578→   578→  // --- Stage 5: Confidence Calculation ---
   579→   579→  callbacks.onProgress('confidence', 'Calculating confidence scores...', 72);
   580→   580→  // Include phone_validation evidence in source count
   581→   581→  const phoneEvCount = allEvidence.filter(e => e.sourceType === 'phone_validation').length;
   582→   582→  const overallConfidence = candidates.length > 0
   583→   583→    ? clampScore(
   584→   584→        candidates[0].confidence * 0.7 +
   585→   585→        Math.min((candidates[0].evidence?.length || 0) * 4, 20) +
   586→   586→        (phoneEvCount > 0 ? 10 : 0)
   587→   587→      )
   588→   588→    : 0;
   589→   589→
   590→   590→  // --- Stage 6: AI Assessment ---
   591→   591→  callbacks.onProgress('ai_analysis', 'Running AI analysis...', 85);
   592→   592→  const baseInvestigation: Investigation = {
   593→   593→    id,
   594→   594→    status: 'completed',
   595→   595→    depth: config.depth,
   596→   596→    isBatch: false,
   597→   597→    batchId: null,
   598→   598→    inputPhone: config.phone || null,
   599→   599→    inputPhoneNormalized: config.phoneNormalized || null,
   600→   600→    inputEmail: config.email || null,
   601→   601→    inputName: candidates[0]?.name || null,
   602→   602→    inputBusiness: null,
   603→   603→    inputRegion: null,
   604→   604→    inputCountry: country || null,
   605→   605→    inputState: null,
   606→   606→    inputCity: phoneResult.phoneInfo?.location ? String(phoneResult.phoneInfo.location) : null,
   607→   607→    summary: `Investigation completed for ${config.phone || config.email || 'unknown identifier'}. ${candidates.length} identity candidate${candidates.length !== 1 ? 's' : ''} found.`,
   608→   608→    identityCount: candidates.length,
   609→   609→    evidenceCount: allEvidence.length,
   610→   610→    sourceCount: new Set(allEvidence.map(e => e.sourceName)).size,
   611→   611→    confidence: overallConfidence,
   612→   612→    hasConflicts: allEvidence.some(e => e.verificationStatus === 'conflicting'),
   613→   613→    locationStatus: 'unavailable',
   614→   614→    isDemoData: false,
   615→   615→    startedAt: now,
   616→   616→    completedAt: timestampNow(),
   617→   617→    createdAt: now,
   618→   618→    updatedAt: timestampNow(),
   619→   619→    candidates,
   620→   620→    evidence: updatedEvidence,
   621→   621→    locations: [],
   622→   622→    timeline: allTimeline,
   623→   623→  };
   624→   624→
   625→   625→  const { aiAssessment, timeline: aiTimeline } = await stageAIAnalysis(baseInvestigation);
   626→   626→  allTimeline.push(...aiTimeline);
   627→   627→  baseInvestigation.timeline = allTimeline;
   628→   628→
   629→   629→  // --- Stage 7: Finalize ---
   630→   630→  callbacks.onProgress('completing', 'Generating report...', 93);
   631→   631→
   632→   632→  if (aiAssessment) {
   633→   633→    // Sanity cap: AI sometimes over-estimates confidence with minimal evidence.
   634→   634→    // If the only evidence source is phone_validation, cap confidence at 45.
   635→   635→    const nonPhoneEvidence = allEvidence.filter(e => e.sourceType !== 'phone_validation');
   636→   636→    const webEvidence = nonPhoneEvidence.filter(e => e.sourceType === 'web_search' && e.relevanceScore >= 60);
   637→   637→    let aiScore = aiAssessment.confidence.score;
   638→   638→    if (nonPhoneEvidence.length === 0 && webEvidence.length === 0) {
   639→   639→      // Only phone validation — cap at LOW range
   640→   640→      aiScore = Math.min(aiScore, 45);
   641→   641→      aiAssessment.confidence.level = 'LOW';
   642→   642→      aiAssessment.confidence.explanation = 'Only phone number validation was available. No web sources or public records linked a specific identity to this number. ' + (aiAssessment.confidence.explanation || '');
   643→   643→    } else if (webEvidence.length < 2) {
   644→   644→      // 0-1 relevant web results — cap at MODERATE
   645→   645→      aiScore = Math.min(aiScore, 65);
   646→   646→      if (aiAssessment.confidence.level === 'HIGH') {
   647→   647→        aiAssessment.confidence.level = 'MODERATE';
   648→   648→      }
   649→   649→    }
   650→   650→    baseInvestigation.confidence = aiScore;
   651→   651→    baseInvestigation.summary = aiAssessment.conclusion;
   652→   652→  }
   653→   653→
   654→   654→  // --- Stage 8: Extract geolocation from NumVerify for globe pins ---
   655→   655→  // NumVerify returns latitude/longitude for validated numbers
   656→   656→  if (phoneResult.phoneInfo && phoneResult.phoneInfo.valid) {
   657→   657→    const lat = parseFloat(phoneResult.phoneInfo.latitude as string);
   658→   658→    const lng = parseFloat(phoneResult.phoneInfo.longitude as string);
   659→   659→    if (!isNaN(lat) && !isNaN(lng)) {
   660→   660→      baseInvestigation.locations.push({
   661→   661→        id: `numverify-${id}`,
   662→   662→        deviceId: null,
   663→   663→        provider: 'NumVerify Geolocation',
   664→   664→        status: 'last_known',
   665→   665→        latitude: lat,
   666→   666→        longitude: lng,
   667→   667→        accuracy: null,
   668→   668→        address: String(phoneResult.phoneInfo.location || phoneResult.phoneInfo.country_name || ''),
   669→   669→        timestamp: timestampNow(),
   670→   670→        freshness: 'recent',
   671→   671→        deviceStatus: null,
   672→   672→        batteryLevel: null,
   673→   673→        networkType: null,
   674→   674→      });
   675→   675→      baseInvestigation.locationStatus = 'last_known';
   676→   676→    }
   677→   677→  }
   678→   678→
   679→   679→  callbacks.onProgress('completed', 'Investigation complete.', 100);
   680→   680→
   681→   681→  return { investigation: baseInvestigation, aiAssessment };
   682→   682→}
