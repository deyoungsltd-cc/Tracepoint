   395→export const useGlobeStore = create<GlobeStore>((set, get) => ({
   396→  markers: [],
   397→  arcs: [],
   398→  focusedLocation: null,
   399→  setMarkers: (markers) => set({ markers }),
   400→  addMarker: (marker) => set((s) => ({ markers: [...s.markers, marker] })),
   401→  clearMarkers: () => set({ markers: [], arcs: [] }),
   402→  setArcs: (arcs) => set({ arcs }),
   403→  focusLocation: (lat, lng) => set({ focusedLocation: { lat, lng } }),
   404→  clearFocus: () => set({ focusedLocation: null }),
   405→
   406→  addInvestigationMarkers: (investigation: Investigation) => {
   407→    const newMarkers: GlobeMarker[] = [];
   408→    const newArcs: GlobeArc[] = [];
   409→
   410→    // Add candidate location markers
   411→    for (const candidate of investigation.candidates) {
   412→      if (candidate.location) {
   413→        // Try to extract a rough lat/lng from location name
   414→        const latlng = roughGeocode(candidate.location);
   415→        if (latlng) {
   416→          newMarkers.push({
   417→            id: `cand-${candidate.id}`,
   418→            lat: latlng.lat,
   419→            lng: latlng.lng,
   420→            label: candidate.name || candidate.location,
   421→            type: 'identity',
   422→            confidence: candidate.confidence,
   423→            investigationId: investigation.id,
   424→          });
   425→        }
   426→      }
   427→    }
   428→
   429→    // Add device location markers
   430→    for (const loc of investigation.locations) {
   431→      if (loc.latitude && loc.longitude) {
   432→        newMarkers.push({
   433→          id: `loc-${loc.id}`,
   434→          lat: loc.latitude,
   435→          lng: loc.longitude,
   436→          label: loc.address || loc.provider,
   437→          type: 'device',
   438→          confidence: undefined,
   439→          investigationId: investigation.id,
   440→        });
   441→      }
   442→    }
   443→
   444→    // Create arcs between markers
   445→    if (newMarkers.length >= 2) {
   446→      for (let i = 0; i < newMarkers.length - 1; i++) {
   447→        newArcs.push({
   448→          id: `arc-${investigation.id}-${i}`,
   449→          startLat: newMarkers[i].lat,
   450→          startLng: newMarkers[i].lng,
   451→          endLat: newMarkers[i + 1].lat,
   452→          endLng: newMarkers[i + 1].lng,
   453→          color: '#c8a24e',
   454→          animated: true,
   455→        });
   456→      }
   457→    }
   458→
   459→    if (newMarkers.length > 0) {
   460→      set((s) => ({
   461→        markers: [...newMarkers, ...s.markers],
   462→        arcs: [...newArcs, ...s.arcs],
   463→        focusedLocation: newMarkers[0] ? { lat: newMarkers[0].lat, lng: newMarkers[0].lng } : s.focusedLocation,
   464→      }));
   465→    }
   466→  },
   467→}));

   654→function generateDemoAIAssessment(inv: Investigation): AIAssessment {
   655→  const primaryCandidate = inv.candidates[0];
   656→  const evidenceCount = inv.evidence.filter((e) => e.candidateId === primaryCandidate?.id).length;
   657→  const conflicting = inv.evidence.filter((e) => e.verificationStatus === 'conflicting');
   658→
   659→  return {
   660→    summary: `I found ${evidenceCount} public sources connecting the provided identifiers to ${primaryCandidate?.name || 'the primary candidate'}. ${evidenceCount >= 2 ? 'Multiple sources agree on the business association and professional identity.' : 'Limited corroborating sources available.'}`,
   661→    conclusion: inv.confidence && inv.confidence >= 80
   662→      ? `IDENTITY VERIFIED — ${inv.confidence}% confidence. ${evidenceCount} independent public sources corroborate the primary identity.`
   663→      : inv.confidence && inv.confidence >= 50
   664→        ? `IDENTITY PARTIALLY VERIFIED — ${inv.confidence}% confidence. Additional sources recommended.`
   665→        : 'INSUFFICIENT EVIDENCE — NO IDENTITY ASSIGNED',
   666→    confidence: {
   667→      score: inv.confidence || 0,
   668→      level: (inv.confidence && inv.confidence >= 80 ? 'HIGH' : inv.confidence && inv.confidence >= 50 ? 'MODERATE' : 'LOW') as AIAssessment['confidence']['level'],
   669→      supportingEvidence: inv.evidence.filter((e) => e.verificationStatus === 'verified').map((e) => e.sourceName),
   670→      conflictingEvidence: conflicting.map((e) => e.sourceName),
   671→      explanation: `Confidence score derived from ${evidenceCount} evidence items across ${inv.sourceCount} independent sources. ${inv.hasConflicts ? 'Conflicting evidence was detected and factored into the score reduction.' : 'No conflicting evidence detected.'}`,
   672→      lastVerified: new Date().toISOString(),
   673→    },
   674→    recommendations: inv.confidence && inv.confidence < 80
   675→      ? ['Cross-reference with official government registry', 'Verify business registration directly', 'Check professional license databases']
   676→      : ['Investigation findings are strong. Consider generating a formal report.'],
   677→    missingEvidence: inv.confidence && inv.confidence < 90
   678→      ? ['Official government ID verification', 'Financial regulatory records', 'Professional license confirmation']
   679→      : [],
   680→  };
