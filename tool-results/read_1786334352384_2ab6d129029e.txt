    38→// --- Phone normalization ---
    39→function normalizePhone(phone: string): string {
    40→  const digits = phone.replace(/\D/g, '');
    41→  if (digits.length === 0) return '';
    42→  if (digits.startsWith('1') && digits.length === 11) return `+${digits}`;
    43→  return `+${digits}`;
    44→}

    67→// --- Confidence meter SVG ---
    68→function ConfidenceMeter({ score }: { score: number }) {
    69→  const radius = 40;
    70→  const circumference = 2 * Math.PI * radius;
    71→  const offset = circumference - (score / 100) * circumference;
    72→  const color =
    73→    score >= 80
    74→      ? '#22c55e'
    75→      : score >= 50
    76→        ? '#f59e0b'
    77→        : '#ef4444';
    78→
    79→  return (
    80→    <div className="relative flex items-center justify-center w-28 h-28">
    81→      <svg className="tp-confidence-ring w-28 h-28" viewBox="0 0 100 100">
    82→        {/* Track */}
    83→        <circle
    84→          cx="50"
    85→          cy="50"
    86→          r={radius}
    87→          fill="none"
    88→          stroke="#2a3322"
    89→          strokeWidth="6"
    90→        />
    91→        {/* Fill */}
    92→        <circle
    93→          cx="50"
    94→          cy="50"
    95→          r={radius}
    96→          fill="none"
    97→          stroke={color}
    98→          strokeWidth="6"
    99→          strokeLinecap="round"
   100→          strokeDasharray={circumference}
   101→          strokeDashoffset={offset}
   102→          className="transition-all duration-1000 ease-out"
   103→        />
   104→      </svg>
   105→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   106→        <span className="text-2xl font-bold" style={{ color }}>
   107→          {score}
   108→        </span>
   109→        <span className="tp-hud text-[0.6rem]">confidence</span>
   110→      </div>
   111→    </div>
   112→  );
   113→}
