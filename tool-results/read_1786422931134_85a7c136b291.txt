    27→// Score helper
    28→function clampScore(val: number, min = 0, max = 100): number {
    29→  return Math.max(min, Math.min(max, Math.round(val)));
    30→}

    32→function evidenceReliability(sourceType: string): number {
    33→  const map: Record<string, number> = {
    34→    government_record: 97,
    35→    official_website: 93,
    36→    business_directory: 80,
    37→    professional_profile: 75,
    38→    news: 85,
    39→    social_profile: 55,
    40→    web_search: 50,
    41→    phone_validation: 95,
    42→    public_record: 88,
    43→    messaging_osint: 40,
    44→  };
    45→  return map[sourceType] || 50;
    46→}

    58→async function proxyNumVerify(phone: string): Promise<Record<string, unknown> | null> {
    59→  try {
    60→    const res = await fetch(`/api/numverify?phone=${encodeURIComponent(phone)}`);
    61→    if (!res.ok) return null;
    62→    return await res.json();
    63→  } catch { return null; }
    64→}

    66→async function proxySerperSearch(query: string): Promise<Array<{ title: string; link: string; snippet: string }>> {
    67→  try {
    68→    const res = await fetch('/api/serper', {
    69→      method: 'POST',
    70→      headers: { 'Content-Type': 'application/json' },
    71→      body: JSON.stringify({ query }),
    72→    });
    73→    if (!res.ok) return [];
    74→    const data = await res.json();
    75→    return (data.organic || []).map((item: Record<string, unknown>) => ({
    76→      title: String(item.title || ''),
    77→      link: String(item.link || ''),
    78→      snippet: String(item.snippet || ''),
    79→    }));
    80→  } catch { return []; }
    81→}

    95→// Stage 1: Validate phone via NumVerify (server proxy) — cached
    96→async function stagePhoneValidation(config: PipelineConfig): Promise<{
    97→  phoneInfo: Record<string, unknown> | null;
    98→  evidence: EvidenceItem[];
    99→  timeline: TimelineEvent[];
   100→  country: string;
   101→}> {
   102→  const evidence: EvidenceItem[] = [];
   103→  const timeline: TimelineEvent[] = [];
   104→  let phoneInfo: Record<string, unknown> | null = null;
   105→  let country = config.country || '';
   106→
   107→  if (config.phoneNormalized) {
   108→    const result = await withCache(
   109→      cacheKey('numverify', config.phoneNormalized),
   110→      () => proxyNumVerify(config.phoneNormalized!),
   111→      TTL.PHONE_VALIDATION
   112→    );
   113→
   114→    if (result && result.valid) {
   115→      phoneInfo = result;
   116→      country = String(result.country_code || country);
   117→      timeline.push({
   118→        id: uuid(),
   119→        eventType: 'phone_validated',
   120→        description: `Phone validated: ${result.line_type || 'Unknown type'}, ${result.carrier || 'Unknown carrier'}, ${result.country_name || ''}`,
   121→        metadata: { valid: true, country: result.country_code, carrier: result.carrier, lineType: result.line_type },
   122→        timestamp: timestampNow(),
   123→      });
   124→      evidence.push({
   125→        id: uuid(),
   126→        claim: `Phone number ${result.international_format} is a valid ${result.line_type || 'phone'} number registered in ${result.country_name || 'unknown'}${result.carrier ? ` via ${result.carrier}` : ''}`,
   127→        sourceName: 'NumVerify Phone Validation',
   128→        sourceType: 'phone_validation',
   129→        sourceUrl: null,
   130→        discoveredAt: timestampNow(),
   131→        publishedAt: null,
   132→        excerpt: `Country: ${result.country_name}, Location: ${result.location || 'N/A'}, Carrier: ${result.carrier || 'N/A'}, Line: ${result.line_type || 'N/A'}`,
   133→        reliabilityScore: 95,
   134→        relevanceScore: 90,
   135→        freshnessScore: 99,
   136→        verificationStatus: 'verified',
   137→      });
   138→    } else if (result) {
   139→      timeline.push({
   140→        id: uuid(),
   141→        eventType: 'phone_invalid',
   142→        description: 'Phone number could not be validated',
   143→        metadata: { valid: false },
   144→        timestamp: timestampNow(),
   145→      });
   146→    } else {
   147→      timeline.push({
   148→        id: uuid(),
   149→        eventType: 'phone_validation_failed',
   150→        description: 'Phone validation service unavailable',
   151→        metadata: null,
   152→        timestamp: timestampNow(),
   153→      });
   154→    }
   155→  }
   156→
   157→  return { phoneInfo, evidence, timeline, country };
   158→}

   160→// Stage 2: Phone Enrichment (AbstractAPI free tier first, Twilio as fallback) — cached
   161→// AbstractAPI: 100 free calls/month — carrier, line type, VOIP detection, connection status
   162→// Twilio: paid — caller name (CNAM), call forwarding, MCC/MNC
   163→async function stagePhoneEnrichment(
   164→  config: PipelineConfig,
   165→  currentCountry: string,
   166→  hasWebCandidates: boolean,
   167→  fallbackCandidateName: string | null,
   168→): Promise<{
   169→  evidence: EvidenceItem[];
   170→  timeline: TimelineEvent[];
   171→  country: string;
   172→  fallbackCandidateName: string | null;
   173→  callerName: string | null;
   174→}> {
   175→  const evidence: EvidenceItem[] = [];
   176→  const timeline: TimelineEvent[] = [];
   177→  let country = currentCountry;
   178→  let callerName: string | null = null;
   179→
   180→  if (!config.phoneNormalized) {
   181→    return { evidence, timeline, country, fallbackCandidateName, callerName };
   182→  }
   183→
   184→  // --- Try AbstractAPI (FREE) first ---
   185→  const abstractResult: AbstractPhoneResult | null = await withCache(
   186→    cacheKey('abstractphone', config.phoneNormalized),
   187→    () => lookupAbstractPhone(config.phoneNormalized!),
   188→    30 * 60 * 1000
   189→  );
   190→
   191→  if (abstractResult && abstractResult.valid) {
   192→    const provider = 'AbstractAPI';
   193→    const phoneType = abstractResult.type || 'unknown';
   194→    const carrier = abstractResult.carrier || 'Unknown';
   195→    const location = abstractResult.location || '';
   196→    const connected = abstractResult.connected;
   197→    const isVoip = phoneType.toLowerCase() === 'voip';
   198→    const isRoaming = abstractResult.roaming === true;
   199→
   200→    timeline.push({
   201→      id: uuid(),
   202→      eventType: 'abstractphone_enrichment',
   203→      description: `${provider} enrichment: ${carrier} (${phoneType})${location ? ` — ${location}` : ''}`,
   204→      metadata: { provider, carrier, type: phoneType, connected, roaming: isRoaming },
   205→      timestamp: timestampNow(),
   206→    });
   207→
   208→    // Carrier evidence
   209→    evidence.push({
   210→      id: uuid(),
   211→      claim: `Phone ${config.phoneNormalized} is carried by ${carrier} (${phoneType})${location ? ` in ${location}` : ''}`,
   212→      sourceName: `${provider} Phone Enrichment`,
   213→      sourceType: 'phone_validation',
   214→      sourceUrl: null,
   215→      discoveredAt: timestampNow(),
   216→      publishedAt: null,
   217→      excerpt: `Carrier: ${carrier}, Type: ${phoneType}, Location: ${location || 'N/A'}, Connected: ${connected ? 'Yes' : 'No'}${isRoaming ? ', Roaming' : ''}${isVoip ? ', VOIP' : ''}`,
   218→      reliabilityScore: 90,
   219→      relevanceScore: 75,
   220→      freshnessScore: 99,
   221→      verificationStatus: 'verified',
   222→    });
   223→
   224→    // VOIP detection — important for fraud assessment
   225→    if (isVoip) {
   226→      evidence.push({
   227→        id: uuid(),
   228→        claim: `Phone ${config.phoneNormalized} is a VOIP number — potentially higher risk for identity verification`,
   229→        sourceName: `${provider} VOIP Detection`,
   230→        sourceType: 'phone_validation',
   231→        sourceUrl: null,
   232→        discoveredAt: timestampNow(),
   233→        publishedAt: null,
   234→        excerpt: `Line type: VOIP (virtual number). VOIP numbers are often used for temporary or anonymous communications.`,
   235→        reliabilityScore: 85,
   236→        relevanceScore: 70,
   237→        freshnessScore: 99,
   238→        verificationStatus: 'verified',
   239→      });
   240→    }
   241→
   242→    // Connection status
   243→    if (connected !== undefined && connected !== null) {
   244→      evidence.push({
   245→        id: uuid(),
   246→        claim: `Phone ${config.phoneNormalized} is currently ${connected ? 'ACTIVE and connected' : 'DISCONNECTED or inactive'}`,
   247→        sourceName: `${provider} Connection Check`,
   248→        sourceType: 'phone_validation',
   249→        sourceUrl: null,
   250→        discoveredAt: timestampNow(),
   251→        publishedAt: null,
   252→        excerpt: `Status: ${connected ? 'Connected' : 'Disconnected'}`,
   253→        reliabilityScore: 80,
   254→        relevanceScore: 65,
   255→        freshnessScore: 99,
   256→        verificationStatus: connected ? 'verified' : 'possible',
   257→      });
   258→    }
   259→
   260→    // Update country from AbstractAPI
   261→    if (abstractResult.country?.code && !country) {
   262→      country = abstractResult.country.code;
   263→    }
   264→    if (location && !country) {
   265→      // Fallback: try to derive country from location string
   266→      const countryNames: Record<string, string> = {
   267→        'united states': 'US', 'united kingdom': 'GB', 'canada': 'CA', 'australia': 'AU',
   268→        'germany': 'DE', 'france': 'FR', 'india': 'IN', 'japan': 'JP', 'brazil': 'BR',
   269→        'nigeria': 'NG', 'south africa': 'ZA', 'mexico': 'MX', 'china': 'CN',
   270→      };
   271→      const mapped = countryNames[location.toLowerCase()];
   272→      if (mapped) country = mapped;
   273→    }
   274→
   275→    // AbstractAPI doesn't provide caller name, so we skip that here.
   276→    // Twilio below can still add it if configured.
   277→  } else {
   278→    timeline.push({
   279→      id: uuid(),
   280→      eventType: 'abstractphone_skipped',
   281→      description: 'AbstractAPI enrichment skipped — service unavailable or not configured',
   282→      metadata: null,
   283→      timestamp: timestampNow(),
   284→    });
   285→  }
   286→
   287→  // --- Try Twilio as fallback/addition (if configured, adds caller name + call forwarding) ---
   288→  const twilioResult: TwilioLookupResult | null = await withCache(
   289→    cacheKey('twilio', config.phoneNormalized),
   290→    () => lookupTwilio(config.phoneNormalized!),
   291→    TTL.TWILIO_LOOKUP
   292→  );
   293→
   294→  if (twilioResult) {
   295→    // Caller Name (CNAM) — only Twilio provides this
   296→    if (twilioResult.caller_name?.caller_name) {
   297→      const name = twilioResult.caller_name.caller_name;
   298→      callerName = name;
   299→      timeline.push({
   300→        id: uuid(),
   301→        eventType: 'twilio_caller_name',
   302→        description: `Caller name retrieved: ${name}`,
   303→        metadata: { firstName: twilioResult.caller_name.first_name, lastName: twilioResult.caller_name.last_name },
   304→        timestamp: timestampNow(),
   305→      });
   306→      evidence.push({
   307→        id: uuid(),
   308→        claim: `Registered caller name for ${config.phoneNormalized} is "${name}"`,
   309→        sourceName: 'Twilio Caller Name',
   310→        sourceType: 'phone_validation',
   311→        sourceUrl: null,
   312→        discoveredAt: timestampNow(),
   313→        publishedAt: null,
   314→        excerpt: `Caller Name: ${name}${twilioResult.caller_name.first_name ? ` (First: ${twilioResult.caller_name.first_name}, Last: ${twilioResult.caller_name.last_name})` : ''}`,
   315→        reliabilityScore: 85,
   316→        relevanceScore: 80,
   317→        freshnessScore: 99,
   318→        verificationStatus: 'possible',
   319→      });
   320→      if (!hasWebCandidates && !fallbackCandidateName) {
   321→        fallbackCandidateName = name;
   322→      }
   323→    }
   324→
   325→    // Call Forwarding — only Twilio provides this
   326→    if (twilioResult.call_forwarding) {
   327→      const enabled = twilioResult.call_forwarding.enabled;
   328→      timeline.push({
   329→        id: uuid(),
   330→        eventType: 'twilio_call_forwarding',
   331→        description: `Call forwarding: ${enabled ? 'ENABLED' : 'disabled'}`,
   332→        metadata: { enabled },
   333→        timestamp: timestampNow(),
   334→      });
   335→      evidence.push({
   336→        id: uuid(),
   337→        claim: `Call forwarding for ${config.phoneNormalized} is ${enabled ? 'enabled — number may be redirected to another device' : 'disabled'}`,
   338→        sourceName: 'Twilio Call Forwarding Check',
   339→        sourceType: 'phone_validation',
   340→        sourceUrl: null,
   341→        discoveredAt: timestampNow(),
   342→        publishedAt: null,
   343→        excerpt: `Call Forwarding: ${enabled ? 'Enabled' : 'Disabled'}`,
   344→        reliabilityScore: 95,
   345→        relevanceScore: 65,
   346→        freshnessScore: 99,
   347→        verificationStatus: 'verified',
   348→      });
   349→    }
   350→
   351→    // MCC-based country fallback from Twilio
   352→    const mcc = twilioResult.carrier?.mobile_country_code || twilioResult.line_type_intelligence?.mobile_country_code;
   353→    if (mcc && !country) {
   354→      const mccCountryMap: Record<string, string> = {
   355→        '310': 'US', '311': 'US', '312': 'US', '313': 'US', '314': 'US', '315': 'US', '316': 'US',
   356→        '234': 'GB', '235': 'GB',
   357→        '208': 'FR', '222': 'IT', '262': 'DE', '214': 'ES',
   358→        '404': 'IN', '405': 'IN', '406': 'IN', '407': 'IN', '408': 'IN', '409': 'IN', '410': 'IN', '411': 'IN', '412': 'IN', '413': 'IN', '414': 'IN', '415': 'IN',
   359→        '730': 'CL', '724': 'BR', '334': 'MX', '732': 'CO',
   360→        '440': 'JP', '450': 'KR', '460': 'CN', '454': 'HK', '525': 'SG',
   361→        '530': 'NZ', '505': 'AU', '302': 'CA',
   362→      };
   363→      const mapped = mccCountryMap[mcc];
   364→      if (mapped) country = mapped;
   365→    }
   366→  }
   367→
   368→  return { evidence, timeline, country, fallbackCandidateName, callerName };
   369→}

   371→// Stage 3: Web search via Serper.dev (server proxy) — cached
   372→async function stageWebSearch(config: PipelineConfig, country: string): Promise<{
   373→  searchResults: Array<{ title: string; link: string; snippet: string }>;
   374→  evidence: EvidenceItem[];
   375→  timeline: TimelineEvent[];
   376→}> {
   377→  const evidence: EvidenceItem[] = [];
   378→  const timeline: TimelineEvent[] = [];
   379→  let searchResults: Array<{ title: string; link: string; snippet: string }> = [];
   380→
   381→  const queries: string[] = [];
   382→  if (config.phoneNormalized) queries.push(`"${config.phoneNormalized}"`);
   383→  if (config.email) queries.push(`"${config.email}"`);
   384→  if (config.phoneNormalized && country) {
   385→    queries.push(`${config.phoneNormalized.replace(/[^0-9+]/g, '')} site:linkedin.com OR site:facebook.com OR site:twitter.com`);
   386→  }
   387→  if (config.phoneNormalized) {
   388→    queries.push(`${config.phoneNormalized.replace(/[^0-9+]/g, '')} site:wa.me OR site:t.me`);
   389→  }
   390→
   391→  const maxQueries = config.depth === 'quick' ? 1 : config.depth === 'standard' ? 3 : queries.length;
   392→
   393→  for (let i = 0; i < Math.min(maxQueries, queries.length); i++) {
   394→    const results = await withCache(
   395→      cacheKey('serper', queries[i]),
   396→      () => proxySerperSearch(queries[i]),
   397→      TTL.WEB_SEARCH
   398→    );
   399→    searchResults.push(...results);
   400→    timeline.push({
   401→      id: uuid(),
   402→      eventType: 'web_search',
   403→      description: `Search "${queries[i].substring(0, 60)}..." returned ${results.length} results`,
   404→      metadata: { query: queries[i], count: results.length },
   405→      timestamp: timestampNow(),
   406→    });
   407→  }
   408→
   409→  for (const result of searchResults) {
   410→    const relScore = clampScore(
   411→      50 + (config.email && result.snippet.toLowerCase().includes(config.email.toLowerCase()) ? 30 : 0) +
   412→      (config.phoneNormalized && result.snippet.toLowerCase().includes(config.phoneNormalized.replace(/[^0-9]/g, '')) ? 25 : 0)
   413→    );
   414→
   415→    evidence.push({
   416→      id: uuid(),
   417→      claim: result.title,
   418→      sourceName: (() => { try { return new URL(result.link).hostname.replace('www.', ''); } catch { return 'unknown'; } })(),
   419→      sourceType: 'web_search',
   420→      sourceUrl: result.link,
   421→      discoveredAt: timestampNow(),
   422→      publishedAt: null,
   423→      excerpt: result.snippet,
   424→      reliabilityScore: evidenceReliability('web_search'),
   425→      relevanceScore: relScore,
   426→      freshnessScore: 50,
   427→      verificationStatus: relScore >= 70 ? 'possible' : 'unverified',
   428→    });
   429→  }
   430→
   431→  return { searchResults, evidence, timeline };
   432→}

   434→// Stage 4 (NEW): Social Profile Scraper
   435→async function stageSocialScraper(
   436→  searchResults: Array<{ title: string; link: string; snippet: string }>,
   437→  config: PipelineConfig,
   438→): Promise<{ evidence: EvidenceItem[]; timeline: TimelineEvent[]; profiles: ScrapedProfile[] }> {
   439→  const evidence: EvidenceItem[] = [];
   440→  const timeline: TimelineEvent[] = [];
   441→  const profiles: ScrapedProfile[] = [];
   442→
   443→  // Only run for 'standard' or 'deep' investigations
   444→  if (config.depth === 'quick') return { evidence, timeline, profiles };
   445→
   446→  const socialUrls = extractSocialUrls(searchResults);
   447→  if (socialUrls.length === 0) {
   448→    return { evidence, timeline, profiles };
   449→  }
   450→
   451→  timeline.push({
   452→    id: uuid(),
   453→    eventType: 'social_scrape_start',
   454→    description: `Scraping ${socialUrls.length} social profile(s)...`,
   455→    metadata: { urls: socialUrls },
   456→    timestamp: timestampNow(),
   457→  });
   458→
   459→  // Scrape social profiles (limited concurrency)
   460→  const maxScrape = config.depth === 'deep' ? 5 : 3;
   461→  const scraped = await scrapeSocialProfiles(socialUrls.slice(0, maxScrape));
   462→  profiles.push(...scraped);
   463→
   464→  for (const profile of scraped) {
   465→    timeline.push({
   466→      id: uuid(),
   467→      eventType: 'social_profile_found',
   468→      description: `${profile.platform} profile found: ${profile.name || 'Unknown'}`,
   469→      metadata: { platform: profile.platform, name: profile.name, followers: profile.followers },
   470→      timestamp: timestampNow(),
   471→    });
   472→
   473→    evidence.push({
   474→      id: uuid(),
   475→      claim: `${profile.platform} profile: ${profile.name || 'Unknown'}${profile.location ? ` — ${profile.location}` : ''}`,
   476→      sourceName: `${profile.platform} Profile`,
   477→      sourceType: 'social_profile',
   478→      sourceUrl: profile.url,
   479→      discoveredAt: timestampNow(),
   480→      publishedAt: null,
   481→      excerpt: profile.bio || profile.rawSnippet || 'No bio available',
   482→      reliabilityScore: evidenceReliability('social_profile'),
   483→      relevanceScore: profile.name ? 65 : 40,
   484→      freshnessScore: 70,
   485→      verificationStatus: 'possible',
   486→    });
   487→
   488→    // Add follower count as supporting evidence
   489→    if (profile.followers && profile.followers > 0) {
   490→      evidence.push({
   491→        id: uuid(),
   492→        claim: `${profile.platform} account has ${profile.followers.toLocaleString()} followers/subscribers`,
   493→        sourceName: `${profile.platform} Metrics`,
   494→        sourceType: 'social_profile',
   495→        sourceUrl: profile.url,
   496→        discoveredAt: timestampNow(),
   497→        publishedAt: null,
   498→        excerpt: `Followers: ${profile.followers.toLocaleString()}`,
   499→        reliabilityScore: 50,
   500→        relevanceScore: 45,
   501→        freshnessScore: 70,
   502→        verificationStatus: 'unverified',
   503→      });
   504→    }
   505→
   506→    // Add username as evidence for Telegram
   507→    if (profile.platform === 'Telegram' && profile.url) {
   508→      const usernameMatch = profile.url.match(/t\.me\/([a-zA-Z0-9_]{5,})/);
   509→      if (usernameMatch && usernameMatch[1] !== 'joinchat') {
   510→        evidence.push({
   511→          id: uuid(),
   512→          claim: `Telegram username: @${usernameMatch[1]}`,
   513→          sourceName: 'Telegram Username',
   514→          sourceType: 'social_profile',
   515→          sourceUrl: profile.url,
   516→          discoveredAt: timestampNow(),
   517→          publishedAt: null,
   518→          excerpt: `Username: @${usernameMatch[1]}`,
   519→          reliabilityScore: 60,
   520→          relevanceScore: 55,
   521→          freshnessScore: 80,
   522→          verificationStatus: 'possible',
   523→        });
   524→      }
   525→    }
   526→  }
   527→
   528→  return { evidence, timeline, profiles };
   529→}

   531→// Stage 5 (NEW): Messaging Platform OSINT
   532→async function stageMessagingOSINT(config: PipelineConfig): Promise<{
   533→  evidence: EvidenceItem[];
   534→  timeline: TimelineEvent[];
   535→  results: MessagingCheckResult[];
   536→}> {
   537→  const evidence: EvidenceItem[] = [];
   538→  const timeline: TimelineEvent[] = [];
   539→
   540→  // Only run for 'deep' investigations with a phone number
   541→  if (config.depth !== 'deep' || !config.phoneNormalized) {
   542→    return { evidence, timeline, results: [] };
   543→  }
   544→
   545→  timeline.push({
   546→    id: uuid(),
   547→    eventType: 'messaging_osint_start',
   548→    description: 'Checking messaging platform registrations (WhatsApp, Telegram, Signal)...',
   549→    metadata: null,
   550→    timestamp: timestampNow(),
   551→  });
   552→
   553→  const results = await checkAllMessagingPlatforms(config.phoneNormalized);
   554→
   555→  for (const check of results) {
   556→    const isRegistered = check.isRegistered === true;
   557→    const isUnknown = check.isRegistered === 'unknown';
   558→
   559→    timeline.push({
   560→      id: uuid(),
   561→      eventType: `messaging_${check.platform.toLowerCase()}`,
   562→      description: `${check.platform}: ${isRegistered ? 'Indicators found' : isUnknown ? 'No public indicators' : 'Not detected'}`,
   563→      metadata: { platform: check.platform, registered: check.isRegistered, indicator: check.indicator },
   564→      timestamp: timestampNow(),
   565→    });
   566→
   567→    if (isRegistered && check.evidence) {
   568→      evidence.push({
   569→        id: uuid(),
   570→        claim: `${check.platform}: ${check.indicator}`,
   571→        sourceName: `${check.platform} OSINT`,
   572→        sourceType: 'messaging_osint',
   573→        sourceUrl: null,
   574→        discoveredAt: timestampNow(),
   575→        publishedAt: null,
   576→        excerpt: check.evidence.substring(0, 300),
   577→        reliabilityScore: evidenceReliability('messaging_osint'),
   578→        relevanceScore: isRegistered ? 60 : 30,
   579→        freshnessScore: 75,
   580→        verificationStatus: 'possible',
   581→      });
   582→    }
   583→
   584→    // Add Telegram username as a high-value finding
   585→    if (check.platform === 'Telegram' && check.username) {
   586→      evidence.push({
   587→        id: uuid(),
   588→        claim: `Telegram username identified: @${check.username}`,
   589→        sourceName: 'Telegram OSINT',
   590→        sourceType: 'social_profile',
   591→        sourceUrl: `https://t.me/${check.username}`,
   592→        discoveredAt: timestampNow(),
   593→        publishedAt: null,
   594→        excerpt: `Username: @${check.username}`,
   595→        reliabilityScore: 55,
   596→        relevanceScore: 65,
   597→        freshnessScore: 80,
   598→        verificationStatus: 'possible',
   599→      });
   600→    }
   601→  }
   602→
   603→  return { evidence, timeline, results };
   604→}

   606→// Stage 6: Correlate candidates from search results
   607→function stageCorrelation(
   608→  evidence: EvidenceItem[],
   609→  phoneInfo: Record<string, unknown> | null,
   610→  config: PipelineConfig
   611→): { candidates: IdentityCandidate[]; updatedEvidence: EvidenceItem[] } {
   612→  const candidateMap = new Map<string, IdentityCandidate>();
   613→
   614→  for (const ev of evidence) {
   615→    if (ev.sourceType === 'phone_validation') continue;
   616→
   617→    let identityKey = '';
   618→    let name: string | null = null;
   619→    const title = ev.claim || '';
   620→    const snippet = ev.excerpt || '';
   621→
   622→    // Name extraction from different source types
   623→    const nameMatch = title.match(/^([A-Z][a-z]+ [A-Z][a-z]+)/);
   624→    if (nameMatch) {
   625→      name = nameMatch[1];
   626→      identityKey = name.toLowerCase();
   627→    } else if (ev.sourceName && !['google', 'bing', 'yahoo'].includes(ev.sourceName.toLowerCase())) {
   628→      identityKey = ev.sourceName;
   629→    }
   630→
   631→    if (!identityKey) continue;
   632→
   633→    if (!candidateMap.has(identityKey)) {
   634→      const matchFields: string[] = [];
   635→      if (config.phoneNormalized && snippet.includes(config.phoneNormalized.replace(/[^0-9]/g, ''))) matchFields.push('phone');
   636→      if (config.email && snippet.toLowerCase().includes(config.email.toLowerCase())) matchFields.push('email');
   637→
   638→      // Check for Telegram username match
   639→      const telegramMatch = snippet.match(/@([a-zA-Z0-9_]{5,})/);
   640→      if (telegramMatch) matchFields.push('telegram_username');
   641→
   642→      candidateMap.set(identityKey, {
   643→        id: uuid(),
   644→        rank: candidateMap.size + 1,
   645→        name,
   646→        phone: config.phoneNormalized || null,
   647→        email: config.email || null,
   648→        business: null,
   649→        website: ev.sourceUrl && ev.sourceUrl.startsWith('http') ? ev.sourceUrl : null,
   650→        location: phoneInfo && phoneInfo.location ? String(phoneInfo.location) : null,
   651→        photoUrl: null,
   652→        confidence: matchFields.length > 0 ? 60 : 30,
   653→        verifiedStatus: matchFields.length >= 2 ? 'possible' : 'unverified',
   654→        matchFields,
   655→        evidence: [],
   656→      });
   657→    }
   658→
   659→    const candidate = candidateMap.get(identityKey)!;
   660→    ev.candidateId = candidate.id;
   661→    candidate.evidence.push(ev);
   662→
   663→    // Enrich candidate with social platform data
   664→    if (ev.sourceType === 'social_profile' && ev.sourceName.includes('LinkedIn') && !candidate.business) {
   665→      // LinkedIn often has company info in bio
   666→      const bioLines = (ev.excerpt || '').split(/[|·,]/);
   667→      for (const line of bioLines) {
   668→        const trimmed = line.trim();
   669→        if (trimmed.length > 3 && trimmed.length < 50 && !trimmed.includes('@') && !/\d{3}/.test(trimmed)) {
   670→          candidate.business = trimmed;
   671→          break;
   672→        }
   673→      }
   674→    }
   675→  }
   676→
   677→  if (candidateMap.size === 0) {
   678→    const phoneEvidence = evidence.filter(e => e.sourceType === 'phone_validation');
   679→    const phoneValid = phoneInfo && phoneInfo.valid;
   680→
   681→    if (phoneValid || config.email) {
   682→      const candidate: IdentityCandidate = {
   683→        id: uuid(),
   684→        rank: 1,
   685→        name: config.email ? config.email.split('@')[0].replace(/[._]/g, ' ').replace(/\b\w/g, c => c.toUpperCase()) : null,
   686→        phone: phoneValid ? String(phoneInfo.international_format || '') : (config.phoneNormalized || null),
   687→        email: config.email || null,
   688→        business: null,
   689→        website: null,
   690→        location: phoneValid ? String(phoneInfo.location || phoneInfo.country_name || '') : null,
   691→        photoUrl: null,
   692→        confidence: phoneValid ? 55 : 35,
   693→        verifiedStatus: phoneValid ? 'possible' : 'unverified',
   694→        matchFields: [...(phoneValid ? ['phone'] : []), ...(config.email ? ['email'] : [])],
   695→        evidence: [],
   696→      };
   697→
   698→      for (const ev of phoneEvidence) {
   699→        ev.candidateId = candidate.id;
   700→        candidate.evidence.push(ev);
   701→      }
   702→
   703→      return { candidates: [candidate], updatedEvidence: evidence };
   704→    }
   705→
   706→    return { candidates: [], updatedEvidence: evidence };
   707→  }
   708→
   709→  const sorted = Array.from(candidateMap.values()).sort((a, b) => {
   710→    const aScore = a.confidence * 0.4 + a.evidence.length * 15;
   711→    const bScore = b.confidence * 0.4 + b.evidence.length * 15;
   712→    return bScore - aScore;
   713→  });
   714→  sorted.forEach((c, i) => { c.rank = i + 1; });
   715→
   716→  return { candidates: sorted, updatedEvidence: evidence };
   717→}

   719→// Stage 7: AI analysis via OpenAI (already proxied) — cached
   720→async function stageAIAnalysis(
   721→  investigation: Investigation,
   722→): Promise<{ aiAssessment: AIAssessment | null; timeline: TimelineEvent[] }> {
   723→  const timeline: TimelineEvent[] = [];
   724→
   725→  try {
   726→    timeline.push({
   727→      id: uuid(),
   728→      eventType: 'ai_analyzing',
   729→      description: 'AI analyst processing evidence and candidates...',
   730→      metadata: null,
   731→      timestamp: timestampNow(),
   732→    });
   733→
   734→    const assessment = await withCache(
   735→      cacheKey('ai', investigation.id, investigation.evidence.length),
   736→      () => analyzeIdentity(
   737→        {
   738→          phone: investigation.inputPhoneNormalized || '',
   739→          email: investigation.inputEmail || '',
   740→          candidates: investigation.candidates,
   741→          evidence: investigation.evidence,
   742→          country: investigation.inputCountry || '',
   743→        },
   744→        { model: 'gpt-4o' }
   745→      ),
   746→      TTL.AI_ANALYSIS
   747→    );
   748→
   749→    if (assessment) {
   750→      timeline.push({
   751→        id: uuid(),
   752→        eventType: 'ai_completed',
   753→        description: `AI assessment complete — ${assessment.confidence.level} confidence (${assessment.confidence.score}%)`,
   754→        metadata: { level: assessment.confidence.level, score: assessment.confidence.score },
   755→        timestamp: timestampNow(),
   756→      });
   757→    }
   758→
   759→    return { aiAssessment: assessment, timeline };
   760→  } catch {
   761→    timeline.push({
   762→      id: uuid(),
   763→      eventType: 'ai_failed',
   764→      description: 'AI analysis failed — continuing without AI assessment',
   765→      metadata: null,
   766→      timestamp: timestampNow(),
   767→    });
   768→    return { aiAssessment: null, timeline };
   769→  }
   770→}

   776→export async function runRealInvestigation(
   777→  config: PipelineConfig,
   778→  callbacks: PipelineCallbacks
   779→): Promise<{
   780→  investigation: Investigation;
   781→  aiAssessment: AIAssessment | null;
   782→}> {
   783→  const id = uuid();
   784→  const now = timestampNow();
   785→  const allEvidence: EvidenceItem[] = [];
   786→  const allTimeline: TimelineEvent[] = [];
   787→  let country = config.country || '';
   788→
   789→  callbacks.onProgress('initializing', 'Initializing investigation...', 3);
   790→  allTimeline.push({ id: uuid(), eventType: 'started', description: 'Investigation initiated', metadata: { phone: config.phone, email: config.email, depth: config.depth }, timestamp: now });
   791→
   792→  // --- Stage 1: Phone Validation (cached) ---
   793→  callbacks.onProgress('validation', 'Validating phone number...', 8);
   794→  const phoneResult = await stagePhoneValidation(config);
   795→  allEvidence.push(...phoneResult.evidence);
   796→  allTimeline.push(...phoneResult.timeline);
   797→  if (phoneResult.country) country = phoneResult.country;
   798→
   799→  // --- Stage 2: Twilio Enrichment (cached) ---
   800→  callbacks.onProgress('enrichment', 'Enriching phone data via Twilio...', 18);
   801→  const twilioResult = await stageTwilioEnrichment(config, country, false, null);
   802→  allEvidence.push(...twilioResult.evidence);
   803→  allTimeline.push(...twilioResult.timeline);
   804→  if (twilioResult.country) country = twilioResult.country;
   805→  const twilioCallerName = twilioResult.callerName;
   806→
   807→  // --- Stage 3: Web Search (cached) ---
   808→  callbacks.onProgress('discovery', 'Searching public sources...', 30);
   809→  const searchResult = await stageWebSearch(config, country);
   810→  allEvidence.push(...searchResult.evidence);
   811→  allTimeline.push(...searchResult.timeline);
   812→
   813→  // --- Stage 4: Social Profile Scraper (NEW) ---
   814→  callbacks.onProgress('social_scraping', 'Scanning social profiles...', 45);
   815→  const socialResult = await stageSocialScraper(searchResult.searchResults, config);
   816→  allEvidence.push(...socialResult.evidence);
   817→  allTimeline.push(...socialResult.timeline);
   818→
   819→  // --- Stage 5: Messaging OSINT (NEW, deep only) ---
   820→  callbacks.onProgress('messaging_check', 'Checking messaging platforms...', 52);
   821→  const messagingResult = await stageMessagingOSINT(config);
   822→  allEvidence.push(...messagingResult.evidence);
   823→  allTimeline.push(...messagingResult.timeline);
   824→
   825→  // --- Stage 6: Identity Correlation ---
   826→  callbacks.onProgress('correlating', 'Correlating identities...', 62);
   827→  const { candidates, updatedEvidence } = stageCorrelation(allEvidence, phoneResult.phoneInfo, config);
   828→
   829→  // Apply Twilio caller name to candidates that have no name yet
   830→  if (twilioCallerName) {
   831→    for (const c of candidates) {
   832→      if (!c.name) c.name = twilioCallerName;
   833→    }
   834→  }
   835→
   836→  for (const candidate of candidates) {
   837→    const candidateEvidence = allEvidence.filter(e => e.candidateId === candidate.id);
   838→    const baseFromFields = candidate.matchFields.length * 15;
   839→    if (candidateEvidence.length > 0) {
   840→      const avgReliability = candidateEvidence.reduce((sum, e) => sum + e.reliabilityScore, 0) / candidateEvidence.length;
   841→      const avgRelevance = candidateEvidence.reduce((sum, e) => sum + e.relevanceScore, 0) / candidateEvidence.length;
   842→      candidate.confidence = clampScore(avgReliability * 0.4 + avgRelevance * 0.4 + Math.min(candidateEvidence.length * 8, 20) + baseFromFields);
   843→    } else {
   844→      candidate.confidence = clampScore(30 + baseFromFields);
   845→    }
   846→    candidate.verifiedStatus = candidate.confidence >= 80 ? 'verified' : candidate.confidence >= 50 ? 'possible' : 'unverified';
   847→  }
   848→
   849→  // --- Stage 7: Confidence Calculation ---
   850→  callbacks.onProgress('confidence', 'Calculating confidence scores...', 72);
   851→  const phoneEvCount = allEvidence.filter(e => e.sourceType === 'phone_validation').length;
   852→  const socialEvCount = allEvidence.filter(e => e.sourceType === 'social_profile').length;
   853→  const messagingEvCount = allEvidence.filter(e => e.sourceType === 'messaging_osint').length;
   854→  const overallConfidence = candidates.length > 0
   855→    ? clampScore(
   856→        candidates[0].confidence * 0.7 +
   857→        Math.min((candidates[0].evidence?.length || 0) * 4, 20) +
   858→        (phoneEvCount > 0 ? 8 : 0) +
   859→        (socialEvCount > 0 ? 5 : 0) +
   860→        (messagingEvCount > 0 ? 3 : 0)
   861→      )
   862→    : 0;
   863→
   864→  // --- Stage 8: AI Assessment (cached) ---
   865→  callbacks.onProgress('ai_analysis', 'Running AI analysis...', 85);
   866→  const baseInvestigation: Investigation = {
   867→    id,
   868→    status: 'completed',
   869→    depth: config.depth,
   870→    isBatch: false,
   871→    batchId: null,
   872→    inputPhone: config.phone || null,
   873→    inputPhoneNormalized: config.phoneNormalized || null,
   874→    inputEmail: config.email || null,
   875→    inputName: candidates[0]?.name || null,
   876→    inputBusiness: null,
   877→    inputRegion: null,
   878→    inputCountry: country || null,
   879→    inputState: null,
   880→    inputCity: phoneResult.phoneInfo?.location ? String(phoneResult.phoneInfo.location) : null,
   881→    summary: `Investigation completed for ${config.phone || config.email || 'unknown identifier'}. ${candidates.length} identity candidate${candidates.length !== 1 ? 's' : ''} found. ${socialResult.profiles.length} social profile${socialResult.profiles.length !== 1 ? 's' : ''} scanned. ${messagingResult.results.length} messaging platform${messagingResult.results.length !== 1 ? 's' : ''} checked.`,
   882→    identityCount: candidates.length,
   883→    evidenceCount: allEvidence.length,
   884→    sourceCount: new Set(allEvidence.map(e => e.sourceName)).size,
   885→    confidence: overallConfidence,
   886→    hasConflicts: allEvidence.some(e => e.verificationStatus === 'conflicting'),
   887→    locationStatus: 'unavailable',
   888→    isDemoData: false,
   889→    startedAt: now,
   890→    completedAt: timestampNow(),
   891→    createdAt: now,
   892→    updatedAt: timestampNow(),
   893→    candidates,
   894→    evidence: updatedEvidence,
   895→    locations: [],
   896→    timeline: allTimeline,
   897→  };
   898→
   899→  const { aiAssessment, timeline: aiTimeline } = await stageAIAnalysis(baseInvestigation);
   900→  allTimeline.push(...aiTimeline);
   901→  baseInvestigation.timeline = allTimeline;
   902→
   903→  // --- Stage 9: Finalize ---
   904→  callbacks.onProgress('completing', 'Generating report...', 93);
   905→
   906→  if (aiAssessment) {
   907→    // Sanity cap: AI sometimes over-estimates confidence with minimal evidence.
   908→    const nonPhoneEvidence = allEvidence.filter(e => e.sourceType !== 'phone_validation');
   909→    const webEvidence = nonPhoneEvidence.filter(e => e.sourceType === 'web_search' && e.relevanceScore >= 60);
   910→    let aiScore = aiAssessment.confidence.score;
   911→    if (nonPhoneEvidence.length === 0 && webEvidence.length === 0) {
   912→      aiScore = Math.min(aiScore, 45);
   913→      aiAssessment.confidence.level = 'LOW';
   914→      aiAssessment.confidence.explanation = 'Only phone number validation was available. No web sources or public records linked a specific identity to this number. ' + (aiAssessment.confidence.explanation || '');
   915→    } else if (webEvidence.length < 2) {
   916→      aiScore = Math.min(aiScore, 65);
   917→      if (aiAssessment.confidence.level === 'HIGH') {
   918→        aiAssessment.confidence.level = 'MODERATE';
   919→      }
   920→    }
   921→    // Bonus: If social profiles found, allow slightly higher confidence
   922→    if (socialEvCount >= 2 && aiScore >= 50) {
   923→      aiScore = Math.min(aiScore + 5, 95);
   924→    }
   925→    baseInvestigation.confidence = aiScore;
   926→    baseInvestigation.summary = aiAssessment.conclusion;
   927→  }
   928→
   929→  // --- Extract geolocation from NumVerify for globe pins ---
   930→  if (phoneResult.phoneInfo && phoneResult.phoneInfo.valid) {
   931→    const lat = parseFloat(phoneResult.phoneInfo.latitude as string);
   932→    const lng = parseFloat(phoneResult.phoneInfo.longitude as string);
   933→    if (!isNaN(lat) && !isNaN(lng)) {
   934→      baseInvestigation.locations.push({
   935→        id: `numverify-${id}`,
   936→        deviceId: null,
   937→        provider: 'NumVerify Geolocation',
   938→        status: 'last_known',
   939→        latitude: lat,
   940→        longitude: lng,
   941→        accuracy: null,
   942→        address: String(phoneResult.phoneInfo.location || phoneResult.phoneInfo.country_name || ''),
   943→        timestamp: timestampNow(),
   944→        freshness: 'recent',
   945→        deviceStatus: null,
   946→        batteryLevel: null,
   947→        networkType: null,
   948→      });
   949→      baseInvestigation.locationStatus = 'last_known';
   950→    }
   951→  }
   952→
   953→  callbacks.onProgress('completed', 'Investigation complete.', 100);
   954→
   955→  return { investigation: baseInvestigation, aiAssessment };
   956→}
