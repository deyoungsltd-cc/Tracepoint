    27→// Score helper
    28→function clampScore(val: number, min = 0, max = 100): number {
    29→  return Math.max(min, Math.min(max, Math.round(val)));
    30→}

    33→  const map: Record<string, number> = {
    34→    government_record: 97,

    36→    business_directory: 80,
    37→    professional_profile: 75,

    41→    phone_validation: 95,

    58→async function proxyNumVerify(phone: string): Promise<Record<string, unknown> | null> {
    59→  try {
    60→    const res = await fetch(`/api/numverify?phone=${encodeURIComponent(phone)}`);
    61→    if (!res.ok) return null;
    62→    return await res.json();
    63→  } catch { return null; }
    64→}
    65→
    66→async function proxySerperSearch(query: string): Promise<Array<{ title: string; link: string; snippet: string }>> {
    67→  try {
    68→    const res = await fetch('/api/serper', {
    69→      method: 'POST',
    70→      headers: { 'Content-Type': 'application/json' },
    71→      body: JSON.stringify({ query }),
    72→    });
    73→    if (!res.ok) return [];
    74→    const data = await res.json();
    75→    return (data.organic || []).map((item: Record<string, unknown>) => ({
    76→      title: String(item.title || ''),
    77→      link: String(item.link || ''),
    78→      snippet: String(item.snippet || ''),
    79→    }));
    80→  } catch { return []; }
    81→}

    95→// Stage 1: Validate phone via NumVerify (server proxy) — cached
    96→async function stagePhoneValidation(config: PipelineConfig): Promise<{
    97→  phoneInfo: Record<string, unknown> | null;
    98→  evidence: EvidenceItem[];
    99→  timeline: TimelineEvent[];
   100→  country: string;
   101→}> {
   102→  const evidence: EvidenceItem[] = [];
   103→  const timeline: TimelineEvent[] = [];
   104→  let phoneInfo: Record<string, unknown> | null = null;
   105→  let country = config.country || '';
   106→
   107→  if (config.phoneNormalized) {
   108→    const result = await withCache(
   109→      cacheKey('numverify', config.phoneNormalized),
   110→      () => proxyNumVerify(config.phoneNormalized!),
   111→      TTL.PHONE_VALIDATION
   112→    );
   113→
   114→    if (result && result.valid) {
   115→      phoneInfo = result;
   116→      country = String(result.country_code || country);
   117→      timeline.push({
   118→        id: uuid(),
   119→        eventType: 'phone_validated',
   120→        description: `Phone validated: ${result.line_type || 'Unknown type'}, ${result.carrier || 'Unknown carrier'}, ${result.country_name || ''}`,
   121→        metadata: { valid: true, country: result.country_code, carrier: result.carrier, lineType: result.line_type },
   122→        timestamp: timestampNow(),
   123→      });
   124→      evidence.push({
   125→        id: uuid(),
   126→        claim: `Phone number ${result.international_format} is a valid ${result.line_type || 'phone'} number registered in ${result.country_name || 'unknown'}${result.carrier ? ` via ${result.carrier}` : ''}`,
   127→        sourceName: 'NumVerify Phone Validation',
   128→        sourceType: 'phone_validation',
   129→        sourceUrl: null,
   130→        discoveredAt: timestampNow(),
   131→        publishedAt: null,
   132→        excerpt: `Country: ${result.country_name}, Location: ${result.location || 'N/A'}, Carrier: ${result.carrier || 'N/A'}, Line: ${result.line_type || 'N/A'}`,
   133→        reliabilityScore: 95,
   134→        relevanceScore: 90,
   135→        freshnessScore: 99,
   136→        verificationStatus: 'verified',
   137→      });
   138→    } else if (result) {
   139→      timeline.push({
   140→        id: uuid(),
   141→        eventType: 'phone_invalid',
   142→        description: 'Phone number could not be validated',
   143→        metadata: { valid: false },
   144→        timestamp: timestampNow(),
   145→      });
   146→    } else {
   147→      timeline.push({
   148→        id: uuid(),
   149→        eventType: 'phone_validation_failed',
   150→        description: 'Phone validation service unavailable',
   151→        metadata: null,
   152→        timestamp: timestampNow(),
   153→      });
   154→    }
   155→  }
   156→
   157→  return { phoneInfo, evidence, timeline, country };
   158→}

   160→// Stage 2: Phone Enrichment (AbstractAPI free tier first, Twilio as fallback) — cached
   161→// AbstractAPI: 100 free calls/month — carrier, line type, VOIP detection, connection status
   162→// Twilio: paid — caller name (CNAM), call forwarding, MCC/MNC
   163→async function stagePhoneEnrichment(
   164→  config: PipelineConfig,
   165→  currentCountry: string,
   166→  hasWebCandidates: boolean,
   167→  fallbackCandidateName: string | null,
   168→): Promise<{
   169→  evidence: EvidenceItem[];
   170→  timeline: TimelineEvent[];
   171→  country: string;
   172→  fallbackCandidateName: string | null;
   173→  callerName: string | null;
   174→}> {
   175→  const evidence: EvidenceItem[] = [];
   176→  const timeline: TimelineEvent[] = [];
   177→  let country = currentCountry;
   178→  let callerName: string | null = null;
   179→
   180→  if (!config.phoneNormalized) {
   181→    return { evidence, timeline, country, fallbackCandidateName, callerName };
   182→  }
   183→
   184→  // --- Try AbstractAPI (FREE) first ---
   185→  const abstractResult: AbstractPhoneResult | null = await withCache(
   186→    cacheKey('abstractphone', config.phoneNormalized),
   187→    () => lookupAbstractPhone(config.phoneNormalized!),
   188→    30 * 60 * 1000
   189→  );
   190→
   191→  if (abstractResult && abstractResult.valid) {
   192→    const provider = 'AbstractAPI';
   193→    const phoneType = abstractResult.type || 'unknown';
   194→    const carrier = abstractResult.carrier || 'Unknown';
   195→    const location = abstractResult.location || '';
   196→    const connected = abstractResult.connected;
   197→    const isVoip = phoneType.toLowerCase() === 'voip';
   198→    const isRoaming = abstractResult.roaming === true;
   199→
   200→    timeline.push({
   201→      id: uuid(),
   202→      eventType: 'abstractphone_enrichment',
   203→      description: `${provider} enrichment: ${carrier} (${phoneType})${location ? ` — ${location}` : ''}`,
   204→      metadata: { provider, carrier, type: phoneType, connected, roaming: isRoaming },
   205→      timestamp: timestampNow(),
   206→    });
   207→
   208→    // Carrier evidence
   209→    evidence.push({
   210→      id: uuid(),
   211→      claim: `Phone ${config.phoneNormalized} is carried by ${carrier} (${phoneType})${location ? ` in ${location}` : ''}`,
   212→      sourceName: `${provider} Phone Enrichment`,
   213→      sourceType: 'phone_validation',
   214→      sourceUrl: null,
   215→      discoveredAt: timestampNow(),
   216→      publishedAt: null,
   217→      excerpt: `Carrier: ${carrier}, Type: ${phoneType}, Location: ${location || 'N/A'}, Connected: ${connected ? 'Yes' : 'No'}${isRoaming ? ', Roaming' : ''}${isVoip ? ', VOIP' : ''}`,
   218→      reliabilityScore: 90,
   219→      relevanceScore: 75,
   220→      freshnessScore: 99,
   221→      verificationStatus: 'verified',
   222→    });
   223→
   224→    // VOIP detection — important for fraud assessment
   225→    if (isVoip) {
   226→      evidence.push({
   227→        id: uuid(),
   228→        claim: `Phone ${config.phoneNormalized} is a VOIP number — potentially higher risk for identity verification`,
   229→        sourceName: `${provider} VOIP Detection`,
   230→        sourceType: 'phone_validation',
   231→        sourceUrl: null,
   232→        discoveredAt: timestampNow(),
   233→        publishedAt: null,
   234→        excerpt: `Line type: VOIP (virtual number). VOIP numbers are often used for temporary or anonymous communications.`,
   235→        reliabilityScore: 85,
   236→        relevanceScore: 70,
   237→        freshnessScore: 99,
   238→        verificationStatus: 'verified',
   239→      });
   240→    }
   241→
   242→    // Connection status
   243→    if (connected !== undefined && connected !== null) {
   244→      evidence.push({
   245→        id: uuid(),
   246→        claim: `Phone ${config.phoneNormalized} is currently ${connected ? 'ACTIVE and connected' : 'DISCONNECTED or inactive'}`,
   247→        sourceName: `${provider} Connection Check`,
   248→        sourceType: 'phone_validation',
   249→        sourceUrl: null,
   250→        discoveredAt: timestampNow(),
   251→        publishedAt: null,
   252→        excerpt: `Status: ${connected ? 'Connected' : 'Disconnected'}`,
   253→        reliabilityScore: 80,
   254→        relevanceScore: 65,
   255→        freshnessScore: 99,
   256→        verificationStatus: connected ? 'verified' : 'possible',
   257→      });
   258→    }
   259→
   260→    // Update country from AbstractAPI
   261→    if (abstractResult.country?.code && !country) {
   262→      country = abstractResult.country.code;
   263→    }
   264→    if (location && !country) {
   265→      // Fallback: try to derive country from location string
   266→      const countryNames: Record<string, string> = {
   267→        'united states': 'US', 'united kingdom': 'GB', 'canada': 'CA', 'australia': 'AU',
   268→        'germany': 'DE', 'france': 'FR', 'india': 'IN', 'japan': 'JP', 'brazil': 'BR',
   269→        'nigeria': 'NG', 'south africa': 'ZA', 'mexico': 'MX', 'china': 'CN',
   270→      };
   271→      const mapped = countryNames[location.toLowerCase()];
   272→      if (mapped) country = mapped;
   273→    }
   274→
   275→    // AbstractAPI doesn't provide caller name on all plans.
   276→    // Phone Intelligence API may return caller_name and risk_score.
   277→    if (abstractResult.caller_name) {
   278→      callerName = abstractResult.caller_name;
   279→      timeline.push({
   280→        id: uuid(),
   281→        eventType: 'abstractphone_caller_name',
   282→        description: `Caller name: ${abstractResult.caller_name}`,
   283→        metadata: { name: abstractResult.caller_name },
   284→        timestamp: timestampNow(),
   285→      });
   286→      evidence.push({
   287→        id: uuid(),
   288→        claim: `Registered caller name for ${config.phoneNormalized} is "${abstractResult.caller_name}"`,
   289→        sourceName: `${provider} Caller Name`,
   290→        sourceType: 'phone_validation',
   291→        sourceUrl: null,
   292→        discoveredAt: timestampNow(),
   293→        publishedAt: null,
   294→        excerpt: `Caller Name: ${abstractResult.caller_name}`,
   295→        reliabilityScore: 80,
   296→        relevanceScore: 75,
   297→        freshnessScore: 99,
   298→        verificationStatus: 'possible',
   299→      });
   300→      if (!hasWebCandidates && !fallbackCandidateName) {
   301→        fallbackCandidateName = abstractResult.caller_name;
   302→      }
   303→    }
   304→
   305→    // Risk score (Phone Intelligence API feature)
   306→    if (abstractResult.risk_score !== undefined && abstractResult.risk_score !== null) {
   307→      const risk = abstractResult.risk_score;
   308→      const riskLevel = risk >= 75 ? 'HIGH' : risk >= 40 ? 'MEDIUM' : 'LOW';
   309→      timeline.push({
   310→        id: uuid(),
   311→        eventType: 'abstractphone_risk',
   312→        description: `Fraud risk score: ${risk}/100 (${riskLevel})`,
   313→        metadata: { score: risk, level: riskLevel },
   314→        timestamp: timestampNow(),
   315→      });
   316→      evidence.push({
   317→        id: uuid(),
   318→        claim: `Phone ${config.phoneNormalized} has a fraud risk score of ${risk}/100 (${riskLevel} risk)`,
   319→        sourceName: `${provider} Risk Analysis`,
   320→        sourceType: 'phone_validation',
   321→        sourceUrl: null,
   322→        discoveredAt: timestampNow(),
   323→        publishedAt: null,
   324→        excerpt: `Risk Score: ${risk}/100. ${risk >= 75 ? 'This number exhibits characteristics commonly associated with fraudulent activity.' : risk >= 40 ? 'This number has some risk indicators.' : 'This number appears to be low risk.'}`,
   325→        reliabilityScore: 80,
   326→        relevanceScore: risk >= 75 ? 80 : risk >= 40 ? 65 : 50,
   327→        freshnessScore: 99,
   328→        verificationStatus: 'verified',
   329→      });
   330→    }
   331→
   332→    // Twilio below can still add call forwarding if configured.
   333→  } else {
   334→    timeline.push({
   335→      id: uuid(),
   336→      eventType: 'abstractphone_skipped',
   337→      description: 'AbstractAPI enrichment skipped — service unavailable or not configured',
   338→      metadata: null,
   339→      timestamp: timestampNow(),
   340→    });
   341→  }
   342→
   343→  // --- Try Twilio as fallback/addition (if configured, adds caller name + call forwarding) ---
   344→  const twilioResult: TwilioLookupResult | null = await withCache(
   345→    cacheKey('twilio', config.phoneNormalized),
   346→    () => lookupTwilio(config.phoneNormalized!),
   347→    TTL.TWILIO_LOOKUP
   348→  );
   349→
   350→  if (twilioResult) {
   351→    // Caller Name (CNAM) — only Twilio provides this
   352→    if (twilioResult.caller_name?.caller_name) {
   353→      const name = twilioResult.caller_name.caller_name;
   354→      callerName = name;
   355→      timeline.push({
   356→        id: uuid(),
   357→        eventType: 'twilio_caller_name',
   358→        description: `Caller name retrieved: ${name}`,
   359→        metadata: { firstName: twilioResult.caller_name.first_name, lastName: twilioResult.caller_name.last_name },
   360→        timestamp: timestampNow(),
   361→      });
   362→      evidence.push({
   363→        id: uuid(),
   364→        claim: `Registered caller name for ${config.phoneNormalized} is "${name}"`,
   365→        sourceName: 'Twilio Caller Name',
   366→        sourceType: 'phone_validation',
   367→        sourceUrl: null,
   368→        discoveredAt: timestampNow(),
   369→        publishedAt: null,
   370→        excerpt: `Caller Name: ${name}${twilioResult.caller_name.first_name ? ` (First: ${twilioResult.caller_name.first_name}, Last: ${twilioResult.caller_name.last_name})` : ''}`,
   371→        reliabilityScore: 85,
   372→        relevanceScore: 80,
   373→        freshnessScore: 99,
   374→        verificationStatus: 'possible',
   375→      });
   376→      if (!hasWebCandidates && !fallbackCandidateName) {
   377→        fallbackCandidateName = name;
   378→      }
   379→    }
   380→
   381→    // Call Forwarding — only Twilio provides this
   382→    if (twilioResult.call_forwarding) {
   383→      const enabled = twilioResult.call_forwarding.enabled;
   384→      timeline.push({
   385→        id: uuid(),
   386→        eventType: 'twilio_call_forwarding',
   387→        description: `Call forwarding: ${enabled ? 'ENABLED' : 'disabled'}`,
   388→        metadata: { enabled },
   389→        timestamp: timestampNow(),
   390→      });
   391→      evidence.push({
   392→        id: uuid(),
   393→        claim: `Call forwarding for ${config.phoneNormalized} is ${enabled ? 'enabled — number may be redirected to another device' : 'disabled'}`,
   394→        sourceName: 'Twilio Call Forwarding Check',
   395→        sourceType: 'phone_validation',
   396→        sourceUrl: null,
   397→        discoveredAt: timestampNow(),
   398→        publishedAt: null,
   399→        excerpt: `Call Forwarding: ${enabled ? 'Enabled' : 'Disabled'}`,
   400→        reliabilityScore: 95,
   401→        relevanceScore: 65,
   402→        freshnessScore: 99,
   403→        verificationStatus: 'verified',
   404→      });
   405→    }
   406→
   407→    // MCC-based country fallback from Twilio
   408→    const mcc = twilioResult.carrier?.mobile_country_code || twilioResult.line_type_intelligence?.mobile_country_code;
   409→    if (mcc && !country) {
   410→      const mccCountryMap: Record<string, string> = {
   411→        '310': 'US', '311': 'US', '312': 'US', '313': 'US', '314': 'US', '315': 'US', '316': 'US',
   412→        '234': 'GB', '235': 'GB',
   413→        '208': 'FR', '222': 'IT', '262': 'DE', '214': 'ES',
   414→        '404': 'IN', '405': 'IN', '406': 'IN', '407': 'IN', '408': 'IN', '409': 'IN', '410': 'IN', '411': 'IN', '412': 'IN', '413': 'IN', '414': 'IN', '415': 'IN',
   415→        '730': 'CL', '724': 'BR', '334': 'MX', '732': 'CO',
   416→        '440': 'JP', '450': 'KR', '460': 'CN', '454': 'HK', '525': 'SG',
   417→        '530': 'NZ', '505': 'AU', '302': 'CA',
   418→      };
   419→      const mapped = mccCountryMap[mcc];
   420→      if (mapped) country = mapped;
   421→    }
   422→  }
   423→
   424→  return { evidence, timeline, country, fallbackCandidateName, callerName };
   425→}

   427→// Stage 3: Web search via Serper.dev (server proxy) — cached
   428→async function stageWebSearch(config: PipelineConfig, country: string): Promise<{
   429→  searchResults: Array<{ title: string; link: string; snippet: string }>;
   430→  evidence: EvidenceItem[];
   431→  timeline: TimelineEvent[];
   432→}> {
   433→  const evidence: EvidenceItem[] = [];
   434→  const timeline: TimelineEvent[] = [];
   435→  let searchResults: Array<{ title: string; link: string; snippet: string }> = [];
   436→
   437→  const queries: string[] = [];
   438→  if (config.phoneNormalized) queries.push(`"${config.phoneNormalized}"`);
   439→  if (config.email) queries.push(`"${config.email}"`);
   440→  if (config.phoneNormalized && country) {
   441→    queries.push(`${config.phoneNormalized.replace(/[^0-9+]/g, '')} site:linkedin.com OR site:facebook.com OR site:twitter.com`);
   442→  }
   443→  if (config.phoneNormalized) {
   444→    queries.push(`${config.phoneNormalized.replace(/[^0-9+]/g, '')} site:wa.me OR site:t.me`);
   445→  }
   446→
   447→  const maxQueries = config.depth === 'quick' ? 1 : config.depth === 'standard' ? 3 : queries.length;
   448→
   449→  for (let i = 0; i < Math.min(maxQueries, queries.length); i++) {
   450→    const results = await withCache(
   451→      cacheKey('serper', queries[i]),
   452→      () => proxySerperSearch(queries[i]),
   453→      TTL.WEB_SEARCH
   454→    );
   455→    searchResults.push(...results);
   456→    timeline.push({
   457→      id: uuid(),
   458→      eventType: 'web_search',
   459→      description: `Search "${queries[i].substring(0, 60)}..." returned ${results.length} results`,
   460→      metadata: { query: queries[i], count: results.length },
   461→      timestamp: timestampNow(),
   462→    });
   463→  }
   464→
   465→  for (const result of searchResults) {
   466→    const relScore = clampScore(
   467→      50 + (config.email && result.snippet.toLowerCase().includes(config.email.toLowerCase()) ? 30 : 0) +
   468→      (config.phoneNormalized && result.snippet.toLowerCase().includes(config.phoneNormalized.replace(/[^0-9]/g, '')) ? 25 : 0)
   469→    );
   470→
   471→    evidence.push({
   472→      id: uuid(),
   473→      claim: result.title,
   474→      sourceName: (() => { try { return new URL(result.link).hostname.replace('www.', ''); } catch { return 'unknown'; } })(),
   475→      sourceType: 'web_search',
   476→      sourceUrl: result.link,
   477→      discoveredAt: timestampNow(),
   478→      publishedAt: null,
   479→      excerpt: result.snippet,
   480→      reliabilityScore: evidenceReliability('web_search'),
   481→      relevanceScore: relScore,
   482→      freshnessScore: 50,
   483→      verificationStatus: relScore >= 70 ? 'possible' : 'unverified',
   484→    });
   485→  }
   486→
   487→  return { searchResults, evidence, timeline };
   488→}

   490→// Stage 4 (NEW): Social Profile Scraper
   491→async function stageSocialScraper(
   492→  searchResults: Array<{ title: string; link: string; snippet: string }>,
   493→  config: PipelineConfig,
   494→): Promise<{ evidence: EvidenceItem[]; timeline: TimelineEvent[]; profiles: ScrapedProfile[] }> {
   495→  const evidence: EvidenceItem[] = [];
   496→  const timeline: TimelineEvent[] = [];
   497→  const profiles: ScrapedProfile[] = [];
   498→
   499→  // Only run for 'standard' or 'deep' investigations
   500→  if (config.depth === 'quick') return { evidence, timeline, profiles };
   501→
   502→  const socialUrls = extractSocialUrls(searchResults);
   503→  if (socialUrls.length === 0) {
   504→    return { evidence, timeline, profiles };
   505→  }
   506→
   507→  timeline.push({
   508→    id: uuid(),
   509→    eventType: 'social_scrape_start',
   510→    description: `Scraping ${socialUrls.length} social profile(s)...`,
   511→    metadata: { urls: socialUrls },
   512→    timestamp: timestampNow(),
   513→  });
   514→
   515→  // Scrape social profiles (limited concurrency)
   516→  const maxScrape = config.depth === 'deep' ? 5 : 3;
   517→  const scraped = await scrapeSocialProfiles(socialUrls.slice(0, maxScrape));
   518→  profiles.push(...scraped);
   519→
   520→  for (const profile of scraped) {
   521→    timeline.push({
   522→      id: uuid(),
   523→      eventType: 'social_profile_found',
   524→      description: `${profile.platform} profile found: ${profile.name || 'Unknown'}`,
   525→      metadata: { platform: profile.platform, name: profile.name, followers: profile.followers },
   526→      timestamp: timestampNow(),
   527→    });
   528→
   529→    evidence.push({
   530→      id: uuid(),
   531→      claim: `${profile.platform} profile: ${profile.name || 'Unknown'}${profile.location ? ` — ${profile.location}` : ''}`,
   532→      sourceName: `${profile.platform} Profile`,
   533→      sourceType: 'social_profile',
   534→      sourceUrl: profile.url,
   535→      discoveredAt: timestampNow(),
   536→      publishedAt: null,
   537→      excerpt: profile.bio || profile.rawSnippet || 'No bio available',
   538→      reliabilityScore: evidenceReliability('social_profile'),
   539→      relevanceScore: profile.name ? 65 : 40,
   540→      freshnessScore: 70,
   541→      verificationStatus: 'possible',
   542→    });
   543→
   544→    // Add follower count as supporting evidence
   545→    if (profile.followers && profile.followers > 0) {

   547→        id: uuid(),
   548→        claim: `${profile.platform} account has ${profile.followers.toLocaleString()} followers/subscribers`,
   549→        sourceName: `${profile.platform} Metrics`,
   550→        sourceType: 'social_profile',
   551→        sourceUrl: profile.url,
   552→        discoveredAt: timestampNow(),
   553→        publishedAt: null,
   554→        excerpt: `Followers: ${profile.followers.toLocaleString()}`,
   555→        reliabilityScore: 50,
   556→        relevanceScore: 45,
   557→        freshnessScore: 70,
   558→        verificationStatus: 'unverified',
   559→      });
   560→    }
   561→
   562→    // Add username as evidence for Telegram
   563→    if (profile.platform === 'Telegram' && profile.url) {
   564→      const usernameMatch = profile.url.match(/t\.me\/([a-zA-Z0-9_]{5,})/);
   565→      if (usernameMatch && usernameMatch[1] !== 'joinchat') {
   566→        evidence.push({
   567→          id: uuid(),
   568→          claim: `Telegram username: @${usernameMatch[1]}`,
   569→          sourceName: 'Telegram Username',
   570→          sourceType: 'social_profile',
   571→          sourceUrl: profile.url,
   572→          discoveredAt: timestampNow(),
   573→          publishedAt: null,
   574→          excerpt: `Username: @${usernameMatch[1]}`,
   575→          reliabilityScore: 60,
   576→          relevanceScore: 55,
   577→          freshnessScore: 80,
   578→          verificationStatus: 'possible',
   579→        });
   580→      }
   581→    }
   582→  }
   583→
   584→  return { evidence, timeline, profiles };

   662→// Stage 6: Correlate candidates from search results
   663→function stageCorrelation(
   664→  evidence: EvidenceItem[],
   665→  phoneInfo: Record<string, unknown> | null,
   666→  config: PipelineConfig
   667→): { candidates: IdentityCandidate[]; updatedEvidence: EvidenceItem[] } {
   668→  const candidateMap = new Map<string, IdentityCandidate>();
   669→
   670→  for (const ev of evidence) {
   671→    if (ev.sourceType === 'phone_validation') continue;
   672→
   673→    let identityKey = '';
   674→    let name: string | null = null;
   675→    const title = ev.claim || '';
   676→    const snippet = ev.excerpt || '';
   677→
   678→    // Name extraction from different source types
   679→    const nameMatch = title.match(/^([A-Z][a-z]+ [A-Z][a-z]+)/);
   680→    if (nameMatch) {
   681→      name = nameMatch[1];
   682→      identityKey = name.toLowerCase();
   683→    } else if (ev.sourceName && !['google', 'bing', 'yahoo'].includes(ev.sourceName.toLowerCase())) {
   684→      identityKey = ev.sourceName;
   685→    }
   686→
   687→    if (!identityKey) continue;
   688→
   689→    if (!candidateMap.has(identityKey)) {
   690→      const matchFields: string[] = [];
   691→      if (config.phoneNormalized && snippet.includes(config.phoneNormalized.replace(/[^0-9]/g, ''))) matchFields.push('phone');
   692→      if (config.email && snippet.toLowerCase().includes(config.email.toLowerCase())) matchFields.push('email');
   693→
   694→      // Check for Telegram username match
   695→      const telegramMatch = snippet.match(/@([a-zA-Z0-9_]{5,})/);
   696→      if (telegramMatch) matchFields.push('telegram_username');
   697→
   698→      candidateMap.set(identityKey, {
   699→        id: uuid(),
   700→        rank: candidateMap.size + 1,
   701→        name,
   702→        phone: config.phoneNormalized || null,
   703→        email: config.email || null,
   704→        business: null,
   705→        website: ev.sourceUrl && ev.sourceUrl.startsWith('http') ? ev.sourceUrl : null,
   706→        location: phoneInfo && phoneInfo.location ? String(phoneInfo.location) : null,
   707→        photoUrl: null,
   708→        confidence: matchFields.length > 0 ? 60 : 30,
   709→        verifiedStatus: matchFields.length >= 2 ? 'possible' : 'unverified',
   710→        matchFields,
   711→        evidence: [],
   712→      });
   713→    }
   714→
   715→    const candidate = candidateMap.get(identityKey)!;
   716→    ev.candidateId = candidate.id;
   717→    candidate.evidence.push(ev);
   718→
   719→    // Enrich candidate with social platform data
   720→    if (ev.sourceType === 'social_profile' && ev.sourceName.includes('LinkedIn') && !candidate.business) {
   721→      // LinkedIn often has company info in bio
   722→      const bioLines = (ev.excerpt || '').split(/[|·,]/);
   723→      for (const line of bioLines) {
   724→        const trimmed = line.trim();
   725→        if (trimmed.length > 3 && trimmed.length < 50 && !trimmed.includes('@') && !/\d{3}/.test(trimmed)) {
   726→          candidate.business = trimmed;
   727→          break;
   728→        }
   729→      }
   730→    }
   731→  }
   732→
   733→  if (candidateMap.size === 0) {
   734→    const phoneEvidence = evidence.filter(e => e.sourceType === 'phone_validation');
   735→    const phoneValid = phoneInfo && phoneInfo.valid;
   736→
   737→    if (phoneValid || config.email) {
   738→      const candidate: IdentityCandidate = {
   739→        id: uuid(),
   740→        rank: 1,
   741→        name: config.email ? config.email.split('@')[0].replace(/[._]/g, ' ').replace(/\b\w/g, c => c.toUpperCase()) : null,
   742→        phone: phoneValid ? String(phoneInfo.international_format || '') : (config.phoneNormalized || null),
   743→        email: config.email || null,
   744→        business: null,
   745→        website: null,
   746→        location: phoneValid ? String(phoneInfo.location || phoneInfo.country_name || '') : null,
   747→        photoUrl: null,
   748→        confidence: phoneValid ? 55 : 35,
   749→        verifiedStatus: phoneValid ? 'possible' : 'unverified',
   750→        matchFields: [...(phoneValid ? ['phone'] : []), ...(config.email ? ['email'] : [])],
   751→        evidence: [],
   752→      };
   753→
   754→      for (const ev of phoneEvidence) {
   755→        ev.candidateId = candidate.id;
   756→        candidate.evidence.push(ev);
   757→      }
   758→
   759→      return { candidates: [candidate], updatedEvidence: evidence };
   760→    }
   761→
   762→    return { candidates: [], updatedEvidence: evidence };
   763→  }
   764→
   765→  const sorted = Array.from(candidateMap.values()).sort((a, b) => {
   766→    const aScore = a.confidence * 0.4 + a.evidence.length * 15;
   767→    const bScore = b.confidence * 0.4 + b.evidence.length * 15;
   768→    return bScore - aScore;
   769→  });
   770→  sorted.forEach((c, i) => { c.rank = i + 1; });
   771→
   772→  return { candidates: sorted, updatedEvidence: evidence };
   773→}

   775→// Stage 7: AI analysis via OpenAI (already proxied) — cached
   776→async function stageAIAnalysis(
   777→  investigation: Investigation,
   778→): Promise<{ aiAssessment: AIAssessment | null; timeline: TimelineEvent[] }> {
   779→  const timeline: TimelineEvent[] = [];
   780→
   781→  try {
   782→    timeline.push({
   783→      id: uuid(),
   784→      eventType: 'ai_analyzing',
   785→      description: 'AI analyst processing evidence and candidates...',
   786→      metadata: null,
   787→      timestamp: timestampNow(),
   788→    });
   789→
   790→    const assessment = await withCache(
   791→      cacheKey('ai', investigation.id, investigation.evidence.length),
   792→      () => analyzeIdentity(
   793→        {
   794→          phone: investigation.inputPhoneNormalized || '',
   795→          email: investigation.inputEmail || '',
   796→          candidates: investigation.candidates,
   797→          evidence: investigation.evidence,
   798→          country: investigation.inputCountry || '',
   799→        },
   800→        { model: 'gpt-4o' }
   801→      ),
   802→      TTL.AI_ANALYSIS
   803→    );
   804→
   805→    if (assessment) {
   806→      timeline.push({
   807→        id: uuid(),
   808→        eventType: 'ai_completed',
   809→        description: `AI assessment complete — ${assessment.confidence.level} confidence (${assessment.confidence.score}%)`,
   810→        metadata: { level: assessment.confidence.level, score: assessment.confidence.score },
   811→        timestamp: timestampNow(),
   812→      });
   813→    }
   814→
   815→    return { aiAssessment: assessment, timeline };
   816→  } catch {
   817→    timeline.push({
   818→      id: uuid(),
   819→      eventType: 'ai_failed',
   820→      description: 'AI analysis failed — continuing without AI assessment',
   821→      metadata: null,
   822→      timestamp: timestampNow(),
   823→    });
   824→    return { aiAssessment: null, timeline };
   825→  }
   826→}

   832→export async function runRealInvestigation(
   833→  config: PipelineConfig,
   834→  callbacks: PipelineCallbacks
   835→): Promise<{
   836→  investigation: Investigation;
   837→  aiAssessment: AIAssessment | null;
   838→}> {
   839→  const id = uuid();
   840→  const now = timestampNow();
   841→  const allEvidence: EvidenceItem[] = [];
   842→  const allTimeline: TimelineEvent[] = [];
   843→  let country = config.country || '';
   844→
   845→  callbacks.onProgress('initializing', 'Initializing investigation...', 3);
   846→  allTimeline.push({ id: uuid(), eventType: 'started', description: 'Investigation initiated', metadata: { phone: config.phone, email: config.email, depth: config.depth }, timestamp: now });
   847→
   848→  // --- Stage 1: Phone Validation (cached) ---
   849→  callbacks.onProgress('validation', 'Validating phone number...', 8);
   850→  const phoneResult = await stagePhoneValidation(config);
   851→  allEvidence.push(...phoneResult.evidence);
   852→  allTimeline.push(...phoneResult.timeline);
   853→  if (phoneResult.country) country = phoneResult.country;
   854→
   855→  // --- Stage 2: Phone Enrichment (cached) ---
   856→  callbacks.onProgress('enrichment', 'Enriching phone data...', 18);
   857→  const enrichResult = await stagePhoneEnrichment(config, country, false, null);
   858→  allEvidence.push(...enrichResult.evidence);
   859→  allTimeline.push(...enrichResult.timeline);
   860→  if (enrichResult.country) country = enrichResult.country;
   861→  const callerName = enrichResult.callerName;
   862→
   863→  // --- Stage 3: Web Search (cached) ---
   864→  callbacks.onProgress('discovery', 'Searching public sources...', 30);
   865→  const searchResult = await stageWebSearch(config, country);
   866→  allEvidence.push(...searchResult.evidence);
   867→  allTimeline.push(...searchResult.timeline);
   868→
   869→  // --- Stage 4: Social Profile Scraper (NEW) ---
   870→  callbacks.onProgress('social_scraping', 'Scanning social profiles...', 45);
   871→  const socialResult = await stageSocialScraper(searchResult.searchResults, config);
   872→  allEvidence.push(...socialResult.evidence);
   873→  allTimeline.push(...socialResult.timeline);
   874→
   875→  // --- Stage 5: Messaging OSINT (NEW, deep only) ---
   876→  callbacks.onProgress('messaging_check', 'Checking messaging platforms...', 52);
   877→  const messagingResult = await stageMessagingOSINT(config);
   878→  allEvidence.push(...messagingResult.evidence);
   879→  allTimeline.push(...messagingResult.timeline);
   880→
   881→  // --- Stage 6: Identity Correlation ---
   882→  callbacks.onProgress('correlating', 'Correlating identities...', 62);
   883→  const { candidates, updatedEvidence } = stageCorrelation(allEvidence, phoneResult.phoneInfo, config);
   884→
   885→  // Apply caller name from phone enrichment to candidates that have no name yet
   886→  if (callerName) {
   887→    for (const c of candidates) {
   888→      if (!c.name) c.name = callerName;
   889→    }
   890→  }
   891→
   892→  for (const candidate of candidates) {
   893→    const candidateEvidence = allEvidence.filter(e => e.candidateId === candidate.id);
   894→    const baseFromFields = candidate.matchFields.length * 15;
   895→    if (candidateEvidence.length > 0) {
   896→      const avgReliability = candidateEvidence.reduce((sum, e) => sum + e.reliabilityScore, 0) / candidateEvidence.length;
   897→      const avgRelevance = candidateEvidence.reduce((sum, e) => sum + e.relevanceScore, 0) / candidateEvidence.length;
   898→      candidate.confidence = clampScore(avgReliability * 0.4 + avgRelevance * 0.4 + Math.min(candidateEvidence.length * 8, 20) + baseFromFields);
   899→    } else {
   900→      candidate.confidence = clampScore(30 + baseFromFields);
   901→    }
   902→    candidate.verifiedStatus = candidate.confidence >= 80 ? 'verified' : candidate.confidence >= 50 ? 'possible' : 'unverified';
   903→  }
   904→
   905→  // --- Stage 7: Confidence Calculation ---
   906→  callbacks.onProgress('confidence', 'Calculating confidence scores...', 72);
   907→  const phoneEvCount = allEvidence.filter(e => e.sourceType === 'phone_validation').length;
   908→  const socialEvCount = allEvidence.filter(e => e.sourceType === 'social_profile').length;
   909→  const messagingEvCount = allEvidence.filter(e => e.sourceType === 'messaging_osint').length;
   910→  const overallConfidence = candidates.length > 0
   911→    ? clampScore(
   912→        candidates[0].confidence * 0.7 +
   913→        Math.min((candidates[0].evidence?.length || 0) * 4, 20) +
   914→        (phoneEvCount > 0 ? 8 : 0) +
   915→        (socialEvCount > 0 ? 5 : 0) +
   916→        (messagingEvCount > 0 ? 3 : 0)
   917→      )
   918→    : 0;
   919→
   920→  // --- Stage 8: AI Assessment (cached) ---
   921→  callbacks.onProgress('ai_analysis', 'Running AI analysis...', 85);
   922→  const baseInvestigation: Investigation = {
   923→    id,
   924→    status: 'completed',
   925→    depth: config.depth,
   926→    isBatch: false,
   927→    batchId: null,
   928→    inputPhone: config.phone || null,
   929→    inputPhoneNormalized: config.phoneNormalized || null,
   930→    inputEmail: config.email || null,
   931→    inputName: candidates[0]?.name || null,
   932→    inputBusiness: null,
   933→    inputRegion: null,
   934→    inputCountry: country || null,
   935→    inputState: null,
   936→    inputCity: phoneResult.phoneInfo?.location ? String(phoneResult.phoneInfo.location) : null,
   937→    summary: `Investigation completed for ${config.phone || config.email || 'unknown identifier'}. ${candidates.length} identity candidate${candidates.length !== 1 ? 's' : ''} found. ${socialResult.profiles.length} social profile${socialResult.profiles.length !== 1 ? 's' : ''} scanned. ${messagingResult.results.length} messaging platform${messagingResult.results.length !== 1 ? 's' : ''} checked.`,
   938→    identityCount: candidates.length,
   939→    evidenceCount: allEvidence.length,
   940→    sourceCount: new Set(allEvidence.map(e => e.sourceName)).size,
   941→    confidence: overallConfidence,
   942→    hasConflicts: allEvidence.some(e => e.verificationStatus === 'conflicting'),
   943→    locationStatus: 'unavailable',
   944→    isDemoData: false,
   945→    startedAt: now,
   946→    completedAt: timestampNow(),
   947→    createdAt: now,
   948→    updatedAt: timestampNow(),
   949→    candidates,
   950→    evidence: updatedEvidence,
   951→    locations: [],
   952→    timeline: allTimeline,
   953→  };
   954→
   955→  const { aiAssessment, timeline: aiTimeline } = await stageAIAnalysis(baseInvestigation);
   956→  allTimeline.push(...aiTimeline);

   958→
   959→  // --- Stage 9: Finalize ---
   960→  callbacks.onProgress('completing', 'Generating report...', 93);
   961→
   962→  if (aiAssessment) {
   963→    // Sanity cap: AI sometimes over-estimates confidence with minimal evidence.
   964→    const nonPhoneEvidence = allEvidence.filter(e => e.sourceType !== 'phone_validation');
   965→    const webEvidence = nonPhoneEvidence.filter(e => e.sourceType === 'web_search' && e.relevanceScore >= 60);
   966→    let aiScore = aiAssessment.confidence.score;
   967→    if (nonPhoneEvidence.length === 0 && webEvidence.length === 0) {
   968→      aiScore = Math.min(aiScore, 45);
   969→      aiAssessment.confidence.level = 'LOW';
   970→      aiAssessment.confidence.explanation = 'Only phone number validation was available. No web sources or public records linked a specific identity to this number. ' + (aiAssessment.confidence.explanation || '');
   971→    } else if (webEvidence.length < 2) {
   972→      aiScore = Math.min(aiScore, 65);
   973→      if (aiAssessment.confidence.level === 'HIGH') {
   974→        aiAssessment.confidence.level = 'MODERATE';
   975→      }
   976→    }
   977→    // Bonus: If social profiles found, allow slightly higher confidence
   978→    if (socialEvCount >= 2 && aiScore >= 50) {
   979→      aiScore = Math.min(aiScore + 5, 95);
   980→    }
   981→    baseInvestigation.confidence = aiScore;
   982→    baseInvestigation.summary = aiAssessment.conclusion;
   983→  }
   984→
   985→  // --- Extract geolocation from NumVerify for globe pins ---
   986→  if (phoneResult.phoneInfo && phoneResult.phoneInfo.valid) {
   987→    const lat = parseFloat(phoneResult.phoneInfo.latitude as string);
   988→    const lng = parseFloat(phoneResult.phoneInfo.longitude as string);
   989→    if (!isNaN(lat) && !isNaN(lng)) {
   990→      baseInvestigation.locations.push({
   991→        id: `numverify-${id}`,
   992→        deviceId: null,
   993→        provider: 'NumVerify Geolocation',
   994→        status: 'last_known',
   995→        latitude: lat,
   996→        longitude: lng,
   997→        accuracy: null,
   998→        address: String(phoneResult.phoneInfo.location || phoneResult.phoneInfo.country_name || ''),
   999→        timestamp: timestampNow(),
  1000→        freshness: 'recent',
  1001→        deviceStatus: null,
  1002→        batteryLevel: null,
  1003→        networkType: null,
  1004→      });
  1005→      baseInvestigation.locationStatus = 'last_known';
  1006→    }
  1007→  }
  1008→
  1009→  callbacks.onProgress('completed', 'Investigation complete.', 100);
  1010→
  1011→  return { investigation: baseInvestigation, aiAssessment };
  1012→}
