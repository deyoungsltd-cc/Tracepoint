   386→export const useGlobeStore = create<GlobeStore>((set, get) => ({
   387→  markers: [],
   388→  arcs: [],
   389→  focusedLocation: null,
   390→  setMarkers: (markers) => set({ markers }),
   391→  addMarker: (marker) => set((s) => ({ markers: [...s.markers, marker] })),
   392→  clearMarkers: () => set({ markers: [], arcs: [] }),
   393→  setArcs: (arcs) => set({ arcs }),
   394→  focusLocation: (lat, lng) => set({ focusedLocation: { lat, lng } }),
   395→  clearFocus: () => set({ focusedLocation: null }),
   396→
   397→  addInvestigationMarkers: (investigation: Investigation) => {
   398→    const newMarkers: GlobeMarker[] = [];
   399→    const newArcs: GlobeArc[] = [];
   400→
   401→    // Add candidate location markers
   402→    for (const candidate of investigation.candidates) {
   403→      if (candidate.location) {
   404→        // Try to extract a rough lat/lng from location name
   405→        const latlng = roughGeocode(candidate.location);
   406→        if (latlng) {
   407→          newMarkers.push({
   408→            id: `cand-${candidate.id}`,
   409→            lat: latlng.lat,
   410→            lng: latlng.lng,
   411→            label: candidate.name || candidate.location,
   412→            type: 'identity',
   413→            confidence: candidate.confidence,
   414→            investigationId: investigation.id,
   415→          });
   416→        }
   417→      }
   418→    }
   419→
   420→    // Add device location markers
   421→    for (const loc of investigation.locations) {
   422→      if (loc.latitude && loc.longitude) {
   423→        newMarkers.push({
   424→          id: `loc-${loc.id}`,
   425→          lat: loc.latitude,
   426→          lng: loc.longitude,
   427→          label: loc.address || loc.provider,
   428→          type: 'device',
   429→          confidence: undefined,
   430→          investigationId: investigation.id,
   431→        });
   432→      }
   433→    }
   434→
   435→    // Create arcs between markers
   436→    if (newMarkers.length >= 2) {
   437→      for (let i = 0; i < newMarkers.length - 1; i++) {
   438→        newArcs.push({
   439→          id: `arc-${investigation.id}-${i}`,
   440→          startLat: newMarkers[i].lat,
   441→          startLng: newMarkers[i].lng,
   442→          endLat: newMarkers[i + 1].lat,
   443→          endLng: newMarkers[i + 1].lng,
   444→          color: '#c8a24e',
   445→          animated: true,
   446→        });
   447→      }
   448→    }
   449→
   450→    if (newMarkers.length > 0) {
   451→      set((s) => ({
   452→        markers: [...newMarkers, ...s.markers],
   453→        arcs: [...newArcs, ...s.arcs],
   454→        focusedLocation: newMarkers[0] ? { lat: newMarkers[0].lat, lng: newMarkers[0].lng } : s.focusedLocation,
   455→      }));
   456→    }
   457→  },
   458→}));

   645→function generateDemoAIAssessment(inv: Investigation): AIAssessment {
   646→  const primaryCandidate = inv.candidates[0];
   647→  const evidenceCount = inv.evidence.filter((e) => e.candidateId === primaryCandidate?.id).length;
   648→  const conflicting = inv.evidence.filter((e) => e.verificationStatus === 'conflicting');
   649→
   650→  return {
   651→    summary: `I found ${evidenceCount} public sources connecting the provided identifiers to ${primaryCandidate?.name || 'the primary candidate'}. ${evidenceCount >= 2 ? 'Multiple sources agree on the business association and professional identity.' : 'Limited corroborating sources available.'}`,
   652→    conclusion: inv.confidence && inv.confidence >= 80
   653→      ? `IDENTITY VERIFIED — ${inv.confidence}% confidence. ${evidenceCount} independent public sources corroborate the primary identity.`
   654→      : inv.confidence && inv.confidence >= 50
   655→        ? `IDENTITY PARTIALLY VERIFIED — ${inv.confidence}% confidence. Additional sources recommended.`
   656→        : 'INSUFFICIENT EVIDENCE — NO IDENTITY ASSIGNED',
   657→    confidence: {
   658→      score: inv.confidence || 0,
   659→      level: (inv.confidence && inv.confidence >= 80 ? 'HIGH' : inv.confidence && inv.confidence >= 50 ? 'MODERATE' : 'LOW') as AIAssessment['confidence']['level'],
   660→      supportingEvidence: inv.evidence.filter((e) => e.verificationStatus === 'verified').map((e) => e.sourceName),
   661→      conflictingEvidence: conflicting.map((e) => e.sourceName),
   662→      explanation: `Confidence score derived from ${evidenceCount} evidence items across ${inv.sourceCount} independent sources. ${inv.hasConflicts ? 'Conflicting evidence was detected and factored into the score reduction.' : 'No conflicting evidence detected.'}`,
   663→      lastVerified: new Date().toISOString(),
   664→    },
   665→    recommendations: inv.confidence && inv.confidence < 80
   666→      ? ['Cross-reference with official government registry', 'Verify business registration directly', 'Check professional license databases']
   667→      : ['Investigation findings are strong. Consider generating a formal report.'],
   668→    missingEvidence: inv.confidence && inv.confidence < 90
   669→      ? ['Official government ID verification', 'Financial regulatory records', 'Professional license confirmation']
   670→      : [],
   671→  };
   672→}
