   388→   388→export const useGlobeStore = create<GlobeStore>((set, get) => ({
   389→   389→  markers: [],
   390→   390→  arcs: [],
   391→   391→  focusedLocation: null,
   392→   392→  setMarkers: (markers) => set({ markers }),
   393→   393→  addMarker: (marker) => set((s) => ({ markers: [...s.markers, marker] })),
   394→   394→  clearMarkers: () => set({ markers: [], arcs: [] }),
   395→   395→  setArcs: (arcs) => set({ arcs }),
   396→   396→  focusLocation: (lat, lng) => set({ focusedLocation: { lat, lng } }),
   397→   397→  clearFocus: () => set({ focusedLocation: null }),
   398→   398→
   399→   399→  addInvestigationMarkers: (investigation: Investigation) => {
   400→   400→    const newMarkers: GlobeMarker[] = [];
   401→   401→    const newArcs: GlobeArc[] = [];
   402→   402→
   403→   403→    // Add candidate location markers
   404→   404→    for (const candidate of investigation.candidates) {
   405→   405→      if (candidate.location) {
   406→   406→        // Try to extract a rough lat/lng from location name
   407→   407→        const latlng = roughGeocode(candidate.location);
   408→   408→        if (latlng) {
   409→   409→          newMarkers.push({
   410→   410→            id: `cand-${candidate.id}`,
   411→   411→            lat: latlng.lat,
   412→   412→            lng: latlng.lng,
   413→   413→            label: candidate.name || candidate.location,
   414→   414→            type: 'identity',
   415→   415→            confidence: candidate.confidence,
   416→   416→            investigationId: investigation.id,
   417→   417→          });
   418→   418→        }
   419→   419→      }
   420→   420→    }
   421→   421→
   422→   422→    // Add device location markers
   423→   423→    for (const loc of investigation.locations) {
   424→   424→      if (loc.latitude && loc.longitude) {
   425→   425→        newMarkers.push({
   426→   426→          id: `loc-${loc.id}`,
   427→   427→          lat: loc.latitude,
   428→   428→          lng: loc.longitude,
   429→   429→          label: loc.address || loc.provider,
   430→   430→          type: 'device',
   431→   431→          confidence: undefined,
   432→   432→          investigationId: investigation.id,
   433→   433→        });
   434→   434→      }
   435→   435→    }
   436→   436→
   437→   437→    // Create arcs between markers
   438→   438→    if (newMarkers.length >= 2) {
   439→   439→      for (let i = 0; i < newMarkers.length - 1; i++) {
   440→   440→        newArcs.push({
   441→   441→          id: `arc-${investigation.id}-${i}`,
   442→   442→          startLat: newMarkers[i].lat,
   443→   443→          startLng: newMarkers[i].lng,
   444→   444→          endLat: newMarkers[i + 1].lat,
   445→   445→          endLng: newMarkers[i + 1].lng,
   446→   446→          color: '#c8a24e',
   447→   447→          animated: true,
   448→   448→        });
   449→   449→      }
   450→   450→    }
   451→   451→
   452→   452→    if (newMarkers.length > 0) {
   453→   453→      set((s) => ({
   454→   454→        markers: [...newMarkers, ...s.markers],
   455→   455→        arcs: [...newArcs, ...s.arcs],
   456→   456→        focusedLocation: newMarkers[0] ? { lat: newMarkers[0].lat, lng: newMarkers[0].lng } : s.focusedLocation,
   457→   457→      }));
   458→   458→    }
   459→   459→  },
   460→   460→}));

   647→   647→function generateDemoAIAssessment(inv: Investigation): AIAssessment {
   648→   648→  const primaryCandidate = inv.candidates[0];
   649→   649→  const evidenceCount = inv.evidence.filter((e) => e.candidateId === primaryCandidate?.id).length;
   650→   650→  const conflicting = inv.evidence.filter((e) => e.verificationStatus === 'conflicting');
   651→   651→
   652→   652→  return {
   653→   653→    summary: `I found ${evidenceCount} public sources connecting the provided identifiers to ${primaryCandidate?.name || 'the primary candidate'}. ${evidenceCount >= 2 ? 'Multiple sources agree on the business association and professional identity.' : 'Limited corroborating sources available.'}`,
   654→   654→    conclusion: inv.confidence && inv.confidence >= 80
   655→   655→      ? `IDENTITY VERIFIED — ${inv.confidence}% confidence. ${evidenceCount} independent public sources corroborate the primary identity.`
   656→   656→      : inv.confidence && inv.confidence >= 50
   657→   657→        ? `IDENTITY PARTIALLY VERIFIED — ${inv.confidence}% confidence. Additional sources recommended.`
   658→   658→        : 'INSUFFICIENT EVIDENCE — NO IDENTITY ASSIGNED',
   659→   659→    confidence: {
   660→   660→      score: inv.confidence || 0,
   661→   661→      level: (inv.confidence && inv.confidence >= 80 ? 'HIGH' : inv.confidence && inv.confidence >= 50 ? 'MODERATE' : 'LOW') as AIAssessment['confidence']['level'],
   662→   662→      supportingEvidence: inv.evidence.filter((e) => e.verificationStatus === 'verified').map((e) => e.sourceName),
   663→   663→      conflictingEvidence: conflicting.map((e) => e.sourceName),
   664→   664→      explanation: `Confidence score derived from ${evidenceCount} evidence items across ${inv.sourceCount} independent sources. ${inv.hasConflicts ? 'Conflicting evidence was detected and factored into the score reduction.' : 'No conflicting evidence detected.'}`,
   665→   665→      lastVerified: new Date().toISOString(),
   666→   666→    },
   667→   667→    recommendations: inv.confidence && inv.confidence < 80
   668→   668→      ? ['Cross-reference with official government registry', 'Verify business registration directly', 'Check professional license databases']
   669→   669→      : ['Investigation findings are strong. Consider generating a formal report.'],
   670→   670→    missingEvidence: inv.confidence && inv.confidence < 90
   671→   671→      ? ['Official government ID verification', 'Financial regulatory records', 'Professional license confirmation']
   672→   672→      : [],
   673→   673→  };
