    64→function formatTimestamp(ts: string | null): string {
    65→  if (!ts) return '—';
    66→  const d = new Date(ts);
    67→  const now = new Date();
    68→  const diffMs = now.getTime() - d.getTime();
    69→  const diffMin = Math.floor(diffMs / 60000);
    70→  const diffHr = Math.floor(diffMs / 3600000);
    71→  const diffDay = Math.floor(diffMs / 86400000);
    72→
    73→  if (diffMin < 1) return 'Just now';
    74→  if (diffMin < 60) return `${diffMin}m ago`;
    75→  if (diffHr < 24) return `${diffHr}h ago`;
    76→  if (diffDay < 7) return `${diffDay}d ago`;
    77→  return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' });
    78→}
    79→
    80→function formatDuration(start: string | null, end: string | null): string {
    81→  if (!start || !end) return '—';
    82→  const ms = new Date(end).getTime() - new Date(start).getTime();
    83→  const sec = Math.round(ms / 1000);
    84→  if (sec < 60) return `${sec}s`;
    85→  const min = Math.floor(sec / 60);
    86→  const remSec = sec % 60;
    87→  return `${min}m ${remSec}s`;
    88→}
    89→
    90→function formatFullDate(ts: string | null): string {
    91→  if (!ts) return '—';
    92→  return new Date(ts).toLocaleString('en-US', {
    93→    month: 'short',
    94→    day: 'numeric',
    95→    year: 'numeric',
    96→    hour: '2-digit',
    97→    minute: '2-digit',
    98→    second: '2-digit',
    99→    hour12: false,
   100→  });
   101→}

   272→function ConfidenceRing({ score, size = 72 }: { score: number; size?: number }) {
   273→  const r = (size - 8) / 2;
   274→  const circ = 2 * Math.PI * r;
   275→  const offset = circ - (score / 100) * circ;
   276→  const color = confidenceColor(score);
   277→
   278→  return (
   279→    <div className="relative" style={{ width: size, height: size }}>
   280→      <svg width={size} height={size} className="tp-confidence-ring">
   281→        <circle
   282→          cx={size / 2}
   283→          cy={size / 2}
   284→          r={r}
   285→          fill="none"
   286→          stroke="#2a3322"
   287→          strokeWidth={4}
   288→        />
   289→        <circle
   290→          cx={size / 2}
   291→          cy={size / 2}
   292→          r={r}
   293→          fill="none"
   294→          stroke={color}
   295→          strokeWidth={4}
   296→          strokeDasharray={circ}
   297→          strokeDashoffset={offset}
   298→          strokeLinecap="round"
   299→          style={{ transition: 'stroke-dashoffset 0.6s ease' }}
   300→        />
   301→      </svg>
   302→      <div className="absolute inset-0 flex items-center justify-center">
   303→        <span
   304→          className="tp-hud-value text-lg font-bold"
   305→          style={{ color }}
   306→        >
   307→          {score}%
   308→        </span>
   309→      </div>
   310→    </div>
   311→  );
   312→}

  1215→function SourcesTab() {

  1219→
  1220→  const sourceMap = useMemo(() => {
  1221→    const map = new Map<string, { name: string; type: SourceCategory; count: number; avgReliability: number; url: string | null }>();
  1222→    for (const ev of evidence) {
  1223→      const existing = map.get(ev.sourceName);

  1233→          url: ev.sourceUrl,
  1234→        });
  1235→      }
